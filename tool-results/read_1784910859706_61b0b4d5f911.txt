   351→   351→      setActive(0);
   352→   352→      // focus input after dialog mounts
   353→   353→      requestAnimationFrame(() => inputRef.current?.focus());
   354→   354→    }
   355→   355→  }, [open]);
   356→   356→
   357→   357→  // Keep active index in range
   358→   358→  React.useEffect(() => {
   359→   359→    if (active >= filtered.length) setActive(0);
   360→   360→  }, [filtered, active]);
   361→   361→
   362→   362→  // Scroll active item into view
   363→   363→  React.useEffect(() => {
   364→   364→    const el = listRef.current?.querySelector<HTMLElement>(`[data-idx="${active}"]`);
   365→   365→    el?.scrollIntoView({ block: "nearest" });
   366→   366→  }, [active]);
   367→   367→
   368→   368→  function execute(a: CmdAction) {
   369→   369→    a.run();
   370→   370→    onOpenChange(false);
   371→   371→  }
   372→   372→
   373→   373→  function onKeyDown(e: React.KeyboardEvent) {
   374→   374→    if (e.key === "ArrowDown") {
   375→   375→      e.preventDefault();
   376→   376→      setActive((i) => (i + 1) % Math.max(filtered.length, 1));
   377→   377→    } else if (e.key === "ArrowUp") {
   378→   378→      e.preventDefault();
   379→   379→      setActive((i) => (i - 1 + filtered.length) % Math.max(filtered.length, 1));
   380→   380→    } else if (e.key === "Enter") {
   381→   381→      e.preventDefault();
   382→   382→      const a = filtered[active];
   383→   383→      if (a) execute(a);
   384→   384→    } else if (e.key === "Escape") {
   385→   385→      onOpenChange(false);
   386→   386→    }
   387→   387→  }
   388→   388→
   389→   389→  // Group filtered actions for rendering
   390→   390→  const groups = React.useMemo(() => {
   391→   391→    const map = new Map<string, CmdAction[]>();
   392→   392→    for (const a of filtered) {
   393→   393→      const arr = map.get(a.group) ?? [];
   394→   394→      arr.push(a);
   395→   395→      map.set(a.group, arr);
   396→   396→    }
   397→   397→    return Array.from(map.entries());
   398→   398→  }, [filtered]);
   399→   399→
   400→   400→  // Flat index lookup for keyboard nav
   401→   401→  let flatIdx = -1;
   402→   402→
   403→   403→  return (
   404→   404→    <Dialog open={open} onOpenChange={onOpenChange}>
   405→   405→      <DialogContent
   406→   406→        showCloseButton={false}
   407→   407→        className="p-0 gap-0 sm:max-w-lg max-h-[80vh] overflow-hidden flex flex-col"
   408→   408→        onKeyDown={onKeyDown}
   409→   409→      >
   410→   410→        <DialogTitle className="sr-only">Paleta de comandos</DialogTitle>
   411→   411→        <DialogDescription className="sr-only">
   412→   412→          Busca acciones rápidas o secciones y navega con el teclado.
   413→   413→        </DialogDescription>
   414→   414→        {/* Search input */}
   415→   415→        <div className="flex items-center gap-2.5 px-4 h-14 border-b border-border/60">
   416→   416→          <Search className="h-4 w-4 text-muted-foreground shrink-0" aria-hidden />
   417→   417→          <input
   418→   418→            ref={inputRef}
   419→   419→            type="search"
   420→   420→            value={query}
   421→   421→            onChange={(e) => { setQuery(e.target.value); setActive(0); }}
   422→   422→            placeholder="Busca una acción o sección…"
   423→   423→            className="flex-1 bg-transparent outline-none text-sm placeholder:text-muted-foreground min-w-0"
   424→   424→            aria-label="Buscar acción"
   425→   425→            autoComplete="off"
   426→   426→            spellCheck={false}
   427→   427→          />
   428→   428→          <kbd className="text-[10px] font-mono px-1.5 py-0.5 rounded border border-border/60 shrink-0">ESC</kbd>
   429→   429→        </div>
   430→   430→        {/* Actions list */}
   431→   431→        <div ref={listRef} className="flex-1 overflow-y-auto rp-scroll-thin p-2 max-h-[60vh]">
   432→   432→          {filtered.length === 0 ? (
   433→   433→            <div className="px-3 py-10 text-center text-sm text-muted-foreground flex flex-col items-center gap-2">
   434→   434→              <Sparkles className="h-5 w-5 opacity-50" />
   435→   435→              Sin resultados para “{query}”.
   436→   436→            </div>
   437→   437→          ) : (
   438→   438→            groups.map(([group, items]) => (
   439→   439→              <div key={group} className="mb-1">
   440→   440→                <div className="px-2 py-1.5 text-[10px] font-mono uppercase tracking-[0.18em] text-muted-foreground">
   441→   441→                  {group}
   442→   442→                </div>
   443→   443→                <ul role="listbox" aria-label={group}>
   444→   444→                  {items.map((a) => {
   445→   445→                    flatIdx += 1;
   446→   446→                    const idx = flatIdx;
   447→   447→                    const isActive = idx === active;
   448→   448→                    return (
   449→   449→                      <li key={a.id} role="option" aria-selected={isActive}>
   450→   450→                        <button
   451→   451→                          data-idx={idx}
   452→   452→                          onMouseMove={() => setActive(idx)}
   453→   453→                          onClick={() => execute(a)}
   454→   454→                          className={cn(
   455→   455→                            "w-full flex items-center gap-3 rounded-md px-2.5 py-2 text-left text-sm transition-colors",
   456→   456→                            isActive ? "bg-[var(--gold)]/12 text-[var(--gold-soft)]" : "hover:bg-foreground/5"
   457→   457→                          )}
   458→   458→                        >
   459→   459→                          <a.icon className={cn("h-4 w-4 shrink-0", isActive ? "text-[var(--gold)]" : "text-muted-foreground")} aria-hidden />
   460→   460→                          <span className="flex-1 min-w-0 truncate font-medium">{a.label}</span>
   461→   461→                          <span className="text-[11px] text-muted-foreground shrink-0 hidden sm:inline">{a.hint}</span>
   462→   462→                          {isActive && (
   463→   463→                            <CornerDownLeft className="h-3.5 w-3.5 text-[var(--gold)] shrink-0" aria-hidden />
   464→   464→                          )}
   465→   465→                        </button>
   466→   466→                      </li>
   467→   467→                    );
   468→   468→                  })}
   469→   469→                </ul>
   470→   470→              </div>
   471→   471→            ))
   472→   472→          )}
   473→   473→        </div>
   474→   474→        {/* Footer hint */}
   475→   475→        <div className="flex items-center justify-between gap-2 px-4 h-10 border-t border-border/60 text-[11px] text-muted-foreground">
   476→   476→          <div className="flex items-center gap-3">
   477→   477→            <span className="flex items-center gap-1">
   478→   478→              <kbd className="font-mono px-1 py-0.5 rounded border border-border/60">↑↓</kbd> navegar
   479→   479→            </span>
   480→   480→            <span className="flex items-center gap-1">
   481→   481→              <kbd className="font-mono px-1 py-0.5 rounded border border-border/60">↵</kbd> seleccionar
   482→   482→            </span>
   483→   483→          </div>
   484→   484→          <span className="font-mono">RestoPanel · ⌘K</span>
   485→   485→        </div>
   486→   486→      </DialogContent>
   487→   487→    </Dialog>
   488→   488→  );
   489→   489→}
