   358→function initials(name: string): string {
   359→  return name
   360→    .split(" ")
   361→    .slice(0, 2)
   362→    .map((p) => p[0])
   363→    .join("")
   364→    .toUpperCase();
   365→}
   366→
   367→function formatEur(n: number): string {
   368→  return n.toLocaleString("es-ES", {
   369→    style: "currency",
   370→    currency: "EUR",
   371→    maximumFractionDigits: 0,
   372→  });
   373→}
   374→
   375→function formatDate(iso: string): string {
   376→  // iso "2025-03-12" → "12 mar 2025"
   377→  const d = new Date(iso + "T00:00:00");
   378→  return d.toLocaleDateString("es-ES", {
   379→    day: "2-digit",
   380→    month: "short",
   381→    year: "numeric",
   382→  });
   383→}

   820→/* ------------------------------------------------------------------ */
   821→
   822→function Stars({ value, size = "sm" }: { value: number; size?: "sm" | "md" }) {
   823→  const dim = size === "md" ? "h-4 w-4" : "h-3 w-3";
   824→  const full = Math.floor(value);
   825→  const half = value - full >= 0.5;
   826→  return (
   827→    <span
   828→      className="inline-flex items-center gap-0.5"
   829→      role="img"
   830→      aria-label={`${value.toFixed(1)} de 5 estrellas`}
   831→    >
   832→      {Array.from({ length: 5 }).map((_, i) => {
   833→        const filled = i < full;
   834→        const isHalf = !filled && half && i === full;
   835→        return (
   836→          <Star
   837→            key={i}
   838→            className={cn(
   839→              dim,
   840→              filled || isHalf
   841→                ? "fill-[var(--gold)] text-[var(--gold)]"
   842→                : "fill-transparent text-muted-foreground/40"
   843→            )}
   844→            aria-hidden
   845→          />
   846→        );
   847→      })}
   848→    </span>
   849→  );
   850→}
