   176→function useAnimatedNumber(value: number, duration = 350) {
   177→  const reduce = useReducedMotion();
   178→  const [display, setDisplay] = React.useState(value);
   179→  const fromRef = React.useRef(value);
   180→  const rafRef = React.useRef<number | null>(null);
   181→
   182→  React.useEffect(() => {
   183→    if (reduce) {
   184→      setDisplay(value);
   185→      return;
   186→    }
   187→    const from = fromRef.current;
   188→    const to = value;
   189→    if (from === to) return;
   190→    const start = performance.now();
   191→    function tick(now: number) {
   192→      const t = Math.min(1, (now - start) / duration);
   193→      const eased = 1 - Math.pow(1 - t, 3);
   194→      setDisplay(Math.round(from + (to - from) * eased));
   195→      if (t < 1) {
   196→        rafRef.current = requestAnimationFrame(tick);
   197→      } else {
   198→        fromRef.current = to;
   199→      }
   200→    }
   201→    rafRef.current = requestAnimationFrame(tick);
   202→    return () => {
   203→      if (rafRef.current) cancelAnimationFrame(rafRef.current);
   204→      fromRef.current = value;
   205→    };
   206→  }, [value, duration, reduce]);
   207→
   208→  return display;
   209→}

   583→        state={paymentDialog}
   584→        order={paymentDialog ? orders[paymentDialog.tableId] ?? null : null}
   585→        tableName={paymentDialog ? tables.find((t) => t.id === paymentDialog.tableId)?.name ?? "" : ""}
   586→        onClose={() => setPaymentDialog(null)}
   587→        onCharge={(method, total, tip) => {
   588→          if (!paymentDialog) return;
   589→          const tableName = tables.find((t) => t.id === paymentDialog.tableId)?.name ?? "";
   590→          setOrders((o) => {
   591→            const next = { ...o };
   592→            delete next[paymentDialog.tableId];
   593→            return next;
   594→          });
   595→          setTables((ts) =>
   596→            ts.map((x) =>
   597→              x.id === paymentDialog.tableId
   598→                ? { ...x, status: "libre", occupants: undefined, openedAt: undefined }
   599→                : x
   600→            )
   601→          );
   602→          setPaymentDialog(null);
   603→          toast({
   604→            title: "Cobro realizado",
   605→            description: `${tableName} · ${formatEUR(total + tip)} vía ${method}${tip > 0 ? ` · propina ${formatEUR(tip)}` : ""}`,
   606→          });
   607→        }}
   608→      />
   609→
   610→      {/* Close till dialog */}
   611→      <CloseTillDialog
   612→        open={closeTillDialog}
   613→        onClose={() => setCloseTillDialog(false)}
   614→        onConfirm={closeTill}
   615→        orders={orders}
   616→      />
   617→    </div>
   618→  );
   619→}
