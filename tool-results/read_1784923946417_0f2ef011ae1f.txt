   344→   344→    const q = query.trim().toLowerCase();
   345→   345→    if (!q) return actions;
   346→   346→    return actions.filter((a) =>
   347→   347→      a.label.toLowerCase().includes(q) || a.hint.toLowerCase().includes(q)
   348→   348→    );
   349→   349→  }, [actions, query]);
   350→   350→
   351→   351→  // Reset state on open
   352→   352→  React.useEffect(() => {
   353→   353→    if (open) {
   354→   354→      setQuery("");
   355→   355→      setActive(0);
   356→   356→      // focus input after dialog mounts
   357→   357→      requestAnimationFrame(() => inputRef.current?.focus());
   358→   358→    }
   359→   359→  }, [open]);
   360→   360→
   361→   361→  // Keep active index in range
   362→   362→  React.useEffect(() => {
   363→   363→    if (active >= filtered.length) setActive(0);
   364→   364→  }, [filtered, active]);
   365→   365→
   366→   366→  // Scroll active item into view
   367→   367→  React.useEffect(() => {
   368→   368→    const el = listRef.current?.querySelector<HTMLElement>(`[data-idx="${active}"]`);
   369→   369→    el?.scrollIntoView({ block: "nearest" });
   370→   370→  }, [active]);
   371→   371→
   372→   372→  function execute(a: CmdAction) {
   373→   373→    a.run();
   374→   374→    onOpenChange(false);
   375→   375→  }
   376→   376→
   377→   377→  function onKeyDown(e: React.KeyboardEvent) {
   378→   378→    if (e.key === "ArrowDown") {
   379→   379→      e.preventDefault();
   380→   380→      setActive((i) => (i + 1) % Math.max(filtered.length, 1));
   381→   381→    } else if (e.key === "ArrowUp") {
   382→   382→      e.preventDefault();
   383→   383→      setActive((i) => (i - 1 + filtered.length) % Math.max(filtered.length, 1));
   384→   384→    } else if (e.key === "Enter") {
   385→   385→      e.preventDefault();
   386→   386→      const a = filtered[active];
   387→   387→      if (a) execute(a);
   388→   388→    } else if (e.key === "Escape") {
   389→   389→      onOpenChange(false);
   390→   390→    }
   391→   391→  }
   392→   392→
   393→   393→  // Group filtered actions for rendering
   394→   394→  const groups = React.useMemo(() => {
   395→   395→    const map = new Map<string, CmdAction[]>();
   396→   396→    for (const a of filtered) {
   397→   397→      const arr = map.get(a.group) ?? [];
   398→   398→      arr.push(a);
   399→   399→      map.set(a.group, arr);
   400→   400→    }
   401→   401→    return Array.from(map.entries());
   402→   402→  }, [filtered]);
   403→   403→
   404→   404→  // Flat index lookup for keyboard nav
   405→   405→  let flatIdx = -1;
   406→   406→
   407→   407→  return (
   408→   408→    <Dialog open={open} onOpenChange={onOpenChange}>
   409→   409→      <DialogContent
   410→   410→        showCloseButton={false}
   411→   411→        className="p-0 gap-0 sm:max-w-lg max-h-[80vh] overflow-hidden flex flex-col"
   412→   412→        onKeyDown={onKeyDown}
   413→   413→      >
   414→   414→        <DialogTitle className="sr-only">Paleta de comandos</DialogTitle>
   415→   415→        <DialogDescription className="sr-only">
   416→   416→          Busca acciones rápidas o secciones y navega con el teclado.
   417→   417→        </DialogDescription>
   418→   418→        {/* Search input */}
   419→   419→        <div className="flex items-center gap-2.5 px-4 h-14 border-b border-border/60">
   420→   420→          <Search className="h-4 w-4 text-muted-foreground shrink-0" aria-hidden />
   421→   421→          <input
   422→   422→            ref={inputRef}
   423→   423→            type="search"
   424→   424→            value={query}
   425→   425→            onChange={(e) => { setQuery(e.target.value); setActive(0); }}
   426→   426→            placeholder="Busca una acción o sección…"
   427→   427→            className="flex-1 bg-transparent outline-none text-sm placeholder:text-muted-foreground min-w-0"
   428→   428→            aria-label="Buscar acción"
   429→   429→            autoComplete="off"
   430→   430→            spellCheck={false}
   431→   431→          />
   432→   432→          <kbd className="text-[10px] font-mono px-1.5 py-0.5 rounded border border-border/60 shrink-0">ESC</kbd>
   433→   433→        </div>
   434→   434→        {/* Actions list */}
   435→   435→        <div ref={listRef} className="flex-1 overflow-y-auto rp-scroll-thin p-2 max-h-[60vh]">
   436→   436→          {filtered.length === 0 ? (
   437→   437→            <div className="px-3 py-10 text-center text-sm text-muted-foreground flex flex-col items-center gap-2">
   438→   438→              <Sparkles className="h-5 w-5 opacity-50" />
   439→   439→              Sin resultados para “{query}”.
   440→   440→            </div>
   441→   441→          ) : (
   442→   442→            groups.map(([group, items]) => (
   443→   443→              <div key={group} className="mb-1">
   444→   444→                <div className="px-2 py-1.5 text-[10px] font-mono uppercase tracking-[0.18em] text-muted-foreground">
   445→   445→                  {group}
   446→   446→                </div>
   447→   447→                <ul role="listbox" aria-label={group}>
   448→   448→                  {items.map((a) => {
   449→   449→                    flatIdx += 1;
   450→   450→                    const idx = flatIdx;
   451→   451→                    const isActive = idx === active;
   452→   452→                    return (
   453→   453→                      <li key={a.id} role="option" aria-selected={isActive}>
   454→   454→                        <button
   455→   455→                          data-idx={idx}
   456→   456→                          onMouseMove={() => setActive(idx)}
   457→   457→                          onClick={() => execute(a)}
   458→   458→                          className={cn(
   459→   459→                            "w-full flex items-center gap-3 rounded-md px-2.5 py-2 text-left text-sm transition-colors",
   460→   460→                            isActive ? "bg-[var(--gold)]/12 text-[var(--gold-soft)]" : "hover:bg-foreground/5"
   461→   461→                          )}
   462→   462→                        >
   463→   463→                          <a.icon className={cn("h-4 w-4 shrink-0", isActive ? "text-[var(--gold)]" : "text-muted-foreground")} aria-hidden />
   464→   464→                          <span className="flex-1 min-w-0 truncate font-medium">{a.label}</span>
   465→   465→                          <span className="text-[11px] text-muted-foreground shrink-0 hidden sm:inline">{a.hint}</span>
   466→   466→                          {isActive && (
   467→   467→                            <CornerDownLeft className="h-3.5 w-3.5 text-[var(--gold)] shrink-0" aria-hidden />
   468→   468→                          )}
   469→   469→                        </button>
   470→   470→                      </li>
   471→   471→                    );
   472→   472→                  })}
   473→   473→                </ul>
   474→   474→              </div>
   475→   475→            ))
   476→   476→          )}
   477→   477→        </div>
   478→   478→        {/* Footer hint */}
   479→   479→        <div className="flex items-center justify-between gap-2 px-4 h-10 border-t border-border/60 text-[11px] text-muted-foreground">
   480→   480→          <div className="flex items-center gap-3">
   481→   481→            <span className="flex items-center gap-1">
   482→   482→              <kbd className="font-mono px-1 py-0.5 rounded border border-border/60">↑↓</kbd> navegar
   483→   483→            </span>
   484→   484→            <span className="flex items-center gap-1">
   485→   485→              <kbd className="font-mono px-1 py-0.5 rounded border border-border/60">↵</kbd> seleccionar
   486→   486→            </span>
   487→   487→          </div>
   488→   488→          <span className="font-mono">RestoPanel · ⌘K</span>
   489→   489→        </div>
   490→   490→      </DialogContent>
   491→   491→    </Dialog>
   492→   492→  );
   493→   493→}

   535→   535→class ErrorBoundary extends React.Component<
   536→   536→  { children: React.ReactNode },
   537→   537→  { hasError: boolean; error: Error | null }

   541→   541→    this.state = { hasError: false, error: null };
   542→   542→  }
   543→   543→  static getDerivedStateFromError(error: Error) {
   544→   544→    return { hasError: true, error };
   545→   545→  }
   546→   546→  componentDidCatch(error: Error) {

   548→   548→  }
   549→   549→  render() {
   550→   550→    if (this.state.hasError) {
   551→   551→      return (
   552→   552→        <div className="rp-glass rounded-xl p-8 text-center max-w-2xl mx-auto">
   553→   553→          <p className="text-sm text-destructive mb-3 font-medium">Error al cargar esta sección</p>
   554→   554→          <p className="text-xs text-muted-foreground mb-4 font-mono break-all">{this.state.error?.message?.slice(0, 300)}</p>
   555→   555→          <button
   556→   556→            onClick={() => this.setState({ hasError: false, error: null })}
   557→   557→            className="rounded-md border border-border/60 px-4 py-2 text-sm hover:bg-foreground/5"
   558→   558→          >
   559→   559→            Reintentar
   560→   560→          </button>
   561→   561→        </div>
   562→   562→      );
   563→   563→    }
   564→   564→    return this.props.children;
   565→   565→  }
   566→   566→}
