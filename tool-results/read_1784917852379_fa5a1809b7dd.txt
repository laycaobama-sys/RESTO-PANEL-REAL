   519→/* ============================================================
   520→   Cost trend — line chart (SVG)
   521→============================================================ */
   522→
   523→function CostTrendChart() {
   524→  const reduce = useReducedMotion();
   525→  const W = 600;
   526→  const H = 160;
   527→  const padX = 8;
   528→  const padY = 16;
   529→  const max = Math.max(...COST_TREND);
   530→  const min = Math.min(...COST_TREND);
   531→  const range = max - min || 1;
   532→  const stepX = (W - padX * 2) / (COST_TREND.length - 1);
   533→  const points = COST_TREND.map((v, i) => {
   534→    const x = padX + i * stepX;
   535→    const y = H - padY - ((v - min) / range) * (H - padY * 2);
   536→    return [x, y] as const;
   537→  });
   538→  const linePath = points.map((p, i) => `${i === 0 ? "M" : "L"} ${p[0].toFixed(1)} ${p[1].toFixed(1)}`).join(" ");
   539→  const areaPath = `${linePath} L ${points[points.length - 1][0].toFixed(1)} ${H - padY} L ${points[0][0].toFixed(1)} ${H - padY} Z`;
   540→  const total = COST_TREND.reduce((a, b) => a + b, 0);
   541→  const totalFmt = total.toFixed(2);
   542→
   543→  return (
   544→    <div className="rp-glass rounded-xl p-4 sm:p-5">
   545→      <div className="flex items-end justify-between gap-3 mb-3">
   546→        <div>
   547→          <div className="text-[11px] font-mono uppercase tracking-wider text-muted-foreground">Tendencia de coste · 30 días</div>
   548→          <div className="mt-1 font-display text-2xl font-light rp-gold-text">€{totalFmt}</div>
   549→        </div>
   550→        <div className="text-right text-[11px] text-muted-foreground">
   551→          <div>Min €{min.toFixed(2)} · Max €{max.toFixed(2)}</div>
   552→          <div className="font-mono">diario · EUR</div>
   553→        </div>
   554→      </div>
   555→      <svg viewBox={`0 0 ${W} ${H}`} className="w-full h-auto" preserveAspectRatio="none" role="img" aria-label="Tendencia de coste diario de IA durante 30 días">
   556→        <defs>
   557→          <linearGradient id="costArea" x1="0" y1="0" x2="0" y2="1">
   558→            <stop offset="0%" stopColor="var(--gold)" stopOpacity="0.35" />
   559→            <stop offset="100%" stopColor="var(--gold)" stopOpacity="0" />
   560→          </linearGradient>
   561→          <linearGradient id="costLine" x1="0" y1="0" x2="1" y2="0">
   562→            <stop offset="0%" stopColor="var(--gold-deep)" />
   563→            <stop offset="100%" stopColor="var(--gold-soft)" />
   564→          </linearGradient>
   565→        </defs>
   566→        {/* grid lines */}
   567→        {[0.25, 0.5, 0.75].map((g) => (
   568→          <line key={g} x1={padX} x2={W - padX} y1={padY + g * (H - padY * 2)} y2={padY + g * (H - padY * 2)} stroke="currentColor" strokeOpacity={0.08} strokeWidth={1} />
   569→        ))}
   570→        <motion.path
   571→          d={areaPath}
   572→          fill="url(#costArea)"
   573→          initial={reduce ? false : { opacity: 0 }}
   574→          animate={{ opacity: 1 }}
   575→          transition={{ duration: 0.6, delay: 0.2 }}
   576→        />
   577→        <motion.path
   578→          d={linePath}
   579→          fill="none"
   580→          stroke="url(#costLine)"
   581→          strokeWidth={2}
   582→          strokeLinecap="round"
   583→          strokeLinejoin="round"
   584→          initial={reduce ? false : { pathLength: 0 }}
   585→          animate={{ pathLength: 1 }}
   586→          transition={{ duration: 0.9, ease: "easeInOut" }}
   587→        />
   588→        <circle cx={points[points.length - 1][0]} cy={points[points.length - 1][1]} r={3} fill="var(--gold-soft)" />
   589→      </svg>
   590→      <div className="mt-2 flex items-center justify-between text-[10px] font-mono uppercase tracking-wider text-muted-foreground">
   591→        <span>Hace 30 días</span>
