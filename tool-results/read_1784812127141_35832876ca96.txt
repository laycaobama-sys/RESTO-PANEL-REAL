   566→    <span
   567→      className={cn(
   568→        "inline-flex items-center gap-1 rounded-md border px-1.5 py-0.5 text-[10px] font-mono uppercase tracking-wider",

   571→    >
   572→      {children}
   573→    </span>
   574→  );
   575→}

   581→function KpiCard({ kpi, index }: { kpi: Kpi; index: number }) {
   582→  const reduce = useReducedMotion();
   583→  const Icon = kpi.icon;
   584→  const accentColor =
   585→    kpi.accent === "gold" ? "var(--gold)" : "var(--teal)";
   586→  return (
   587→    <motion.div
   588→      initial={reduce ? false : { opacity: 0, y: 8 }}
