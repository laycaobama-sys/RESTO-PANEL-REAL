    73→/* ============================================================
    74→   Hook: prefers-reduced-motion
    75→============================================================ */
    76→function usePrefersReducedMotion() {
    77→  const [reduce, setReduce] = React.useState(false);
    78→  React.useEffect(() => {
    79→    const mq = window.matchMedia("(prefers-reduced-motion: reduce)");
    80→    setReduce(mq.matches);
    81→    const onChange = () => setReduce(mq.matches);
    82→    mq.addEventListener("change", onChange);
    83→    return () => mq.removeEventListener("change", onChange);
    84→  }, []);
    85→  return reduce;
    86→}

    88→/* ============================================================
    89→   Hook: useLocalStorage
    90→============================================================ */
    91→function useLocalStorage<T>(key: string, initial: T): [T, (v: T | ((p: T) => T)) => void] {
    92→  const [value, setValue] = React.useState<T>(() => {
    93→    if (typeof window === "undefined") return initial;
    94→    try {
    95→      const stored = window.localStorage.getItem(key);
    96→      return stored ? (JSON.parse(stored) as T) : initial;
    97→    } catch {
    98→      return initial;
    99→    }
   100→  });
   101→  React.useEffect(() => {
   102→    try {
   103→      window.localStorage.setItem(key, JSON.stringify(value));
   104→    } catch {
   105→      /* noop */
   106→    }
   107→  }, [key, value]);
   108→  return [value, setValue];
   109→}

  1124→function PasswordInput({
  1125→  label,
  1126→  value,
  1127→  onChange,
  1128→  error,
  1129→  autoComplete,
  1130→  showStrength,
  1131→}: {
  1132→  label: string;
  1133→  value: string;
  1134→  onChange: (v: string) => void;
  1135→  error?: string | null;
  1136→  autoComplete?: string;
  1137→  showStrength?: boolean;
  1138→}) {
  1139→  const [show, setShow] = React.useState(false);
  1140→  const strength = React.useMemo(() => {
  1141→    if (!showStrength) return null;
  1142→    let score = 0;
  1143→    if (value.length >= 8) score++;
  1144→    if (/[A-Z]/.test(value)) score++;
  1145→    if (/[a-z]/.test(value)) score++;
  1146→    if (/[0-9]/.test(value)) score++;
  1147→    if (/[^A-Za-z0-9]/.test(value)) score++;
  1148→    return score;
  1149→  }, [value, showStrength]);
  1150→
  1151→  const strengthLabel = strength === null ? null : strength <= 1 ? "Débil" : strength <= 3 ? "Media" : "Fuerte";
  1152→  const strengthCls = strength === null ? "" : strength <= 1 ? "bg-rose-500" : strength <= 3 ? "bg-amber-400" : "bg-[var(--teal)]";
  1153→
  1154→  return (
  1155→    <div>
  1156→      <Label className="mb-1.5 block text-xs">{label}</Label>
  1157→      <div className="relative">
  1158→        <Lock className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground pointer-events-none" aria-hidden />
  1159→        <input
  1160→          type={show ? "text" : "password"}
  1161→          value={value}
  1162→          onChange={(e) => onChange(e.target.value)}
  1163→          autoComplete={autoComplete}
  1164→          className={cn(
  1165→            "w-full h-11 rounded-md border bg-background/40 pl-10 pr-10 text-sm outline-none transition-colors",
  1166→            error ? "border-rose-400/60 focus:border-rose-400" : "border-border/60 focus:border-[var(--gold)]/50",
  1167→          )}
  1168→          aria-invalid={!!error}
  1169→        />
  1170→        <button
  1171→          type="button"
  1172→          onClick={() => setShow((v) => !v)}
  1173→          className="absolute right-2 top-1/2 -translate-y-1/2 h-8 w-8 flex items-center justify-center rounded text-muted-foreground hover:text-foreground transition-colors"
  1174→          aria-label={show ? "Ocultar contraseña" : "Mostrar contraseña"}
  1175→          tabIndex={-1}
  1176→        >
  1177→          {show ? <EyeOff className="h-4 w-4" /> : <Eye className="h-4 w-4" />}
  1178→        </button>
  1179→      </div>
  1180→      {showStrength && strength !== null && value.length > 0 && (
  1181→        <div className="mt-1.5 flex items-center gap-2">
  1182→          <div className="flex-1 h-1 rounded-full bg-foreground/10 overflow-hidden">
  1183→            <div
  1184→              className={cn("h-full rounded-full transition-all", strengthCls)}
  1185→              style={{ width: `${(strength / 5) * 100}%` }}
  1186→            />
  1187→          </div>
  1188→          <span className="text-[10px] font-mono text-muted-foreground">{strengthLabel}</span>
  1189→        </div>
  1190→      )}
  1191→      <FieldError msg={error} />
  1192→    </div>
  1193→  );
  1194→}
