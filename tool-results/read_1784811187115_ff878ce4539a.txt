   368→
   369→function NewSegmentDialog({
   370→  open,
   371→  onOpenChange,
   372→  onCreate,
   373→}: {
   374→  open: boolean;
   375→  onOpenChange: (v: boolean) => void;
   376→  onCreate: (s: Omit<Segment, "id" | "count">) => void;
   377→}) {
   378→  const [name, setName] = React.useState("");
   379→  const [field, setField] = React.useState<SegmentField>("visitas");
   380→  const [operator, setOperator] = React.useState<SegmentOperator>("gte");
   381→  const [value, setValue] = React.useState("");
   382→  const [errors, setErrors] = React.useState<{ name?: string; value?: string }>({});
   383→
   384→  React.useEffect(() => {
   385→    if (open) {
   386→      setName("");
   387→      setField("visitas");
   388→      setOperator("gte");
   389→      setValue("");
   390→      setErrors({});
   391→    }
   392→  }, [open]);
   393→
   394→  function handleSubmit(ev: React.FormEvent) {
   395→    ev.preventDefault();
   396→    const e: typeof errors = {};
   397→    if (!name.trim()) e.name = "El nombre del segmento es obligatorio.";
   398→    if (!value.trim()) e.value = "El valor de la regla es obligatorio.";
   399→    setErrors(e);
   400→    if (Object.keys(e).length > 0) return;
   401→
   402→    const rule = `${FIELD_LABELS[field]} ${OPERATOR_LABELS[operator]} ${value}`;
   403→    onCreate({ name: name.trim(), rule, field, operator, value: value.trim() });
   404→    onOpenChange(false);
   405→  }
   406→
   407→  return (
   408→    <Dialog open={open} onOpenChange={onOpenChange}>
   409→      <DialogContent className="sm:max-w-lg">
   410→        <DialogHeader>
   411→          <DialogTitle>Nuevo segmento</DialogTitle>
   412→          <DialogDescription>
   413→            Define una regla simple. Las reglas avanzadas (AND/OR, anidadas) están disponibles en el builder completo.
   414→          </DialogDescription>
   415→        </DialogHeader>
   416→        <form onSubmit={handleSubmit} className="space-y-4" noValidate>
   417→          <div className="space-y-2">
   418→            <label htmlFor="seg-name" className="text-xs font-mono uppercase tracking-wider text-muted-foreground">
   419→              Nombre del segmento <span className="text-destructive">*</span>
   420→            </label>
   421→            <Input
   422→              id="seg-name"
   423→              value={name}
   424→              onChange={(e) => setName(e.target.value)}
   425→              placeholder="Ej. Clientes alto LTV"
   426→              aria-invalid={!!errors.name}
   427→              aria-describedby={errors.name ? "seg-name-err" : undefined}
   428→            />
   429→            {errors.name && (
   430→              <p id="seg-name-err" className="text-[11px] text-destructive" role="alert">
   431→                {errors.name}
   432→              </p>
   433→            )}
   434→          </div>
   435→
   436→          <div className="rounded-lg border border-border/60 bg-foreground/[0.02] p-3">
   437→            <div className="mb-2 flex items-center gap-2 text-[11px] font-mono uppercase tracking-wider text-muted-foreground">
   438→              <Filter className="h-3 w-3 text-[var(--teal)]" aria-hidden />
   439→              Regla
   440→            </div>
   441→            <div className="grid grid-cols-1 gap-2 sm:grid-cols-3">
   442→              <div className="space-y-1">
   443→                <label htmlFor="seg-field" className="text-[11px] text-muted-foreground">Campo</label>
   444→                <Select value={field} onValueChange={(v) => setField(v as SegmentField)}>
   445→                  <SelectTrigger id="seg-field" aria-label="Campo de la regla">
   446→                    <SelectValue />
   447→                  </SelectTrigger>
   448→                  <SelectContent>
   449→                    {(Object.keys(FIELD_LABELS) as SegmentField[]).map((f) => (
   450→                      <SelectItem key={f} value={f}>{FIELD_LABELS[f]}</SelectItem>
   451→                    ))}
   452→                  </SelectContent>
   453→                </Select>
   454→              </div>
   455→              <div className="space-y-1">
   456→                <label htmlFor="seg-op" className="text-[11px] text-muted-foreground">Operador</label>
   457→                <Select value={operator} onValueChange={(v) => setOperator(v as SegmentOperator)}>
   458→                  <SelectTrigger id="seg-op" aria-label="Operador de la regla">
   459→                    <SelectValue />
   460→                  </SelectTrigger>
   461→                  <SelectContent>
   462→                    {(Object.keys(OPERATOR_LABELS) as SegmentOperator[]).map((o) => (
   463→                      <SelectItem key={o} value={o}>{OPERATOR_LABELS[o]}</SelectItem>
   464→                    ))}
   465→                  </SelectContent>
   466→                </Select>
   467→              </div>
   468→              <div className="space-y-1">
   469→                <label htmlFor="seg-val" className="text-[11px] text-muted-foreground">
   470→                  Valor <span className="text-destructive">*</span>
   471→                </label>
   472→                <Input
   473→                  id="seg-val"
   474→                  value={value}
   475→                  onChange={(e) => setValue(e.target.value)}
   476→                  placeholder="Ej. 100"
   477→                  aria-invalid={!!errors.value}
   478→                  aria-describedby={errors.value ? "seg-val-err" : undefined}
   479→                />
   480→              </div>
   481→            </div>
   482→            {errors.value && (
   483→              <p id="seg-val-err" className="mt-2 text-[11px] text-destructive" role="alert">
   484→                {errors.value}
   485→              </p>
   486→            )}
   487→            <p className="mt-2 text-[11px] text-muted-foreground">
   488→              Vista previa:{" "}
   489→              <span className="font-mono text-foreground/80">
   490→                {FIELD_LABELS[field]} {OPERATOR_LABELS[operator]} {value || "…"}
   491→              </span>
   492→            </p>
   493→          </div>
   494→
   495→          <DialogFooter>
   496→            <Button type="button" variant="ghost" onClick={() => onOpenChange(false)}>
   497→              Cancelar
   498→            </Button>
   499→            <Button type="submit">
   500→              <Plus className="h-4 w-4" aria-hidden />
   501→              Crear segmento
   502→            </Button>
   503→          </DialogFooter>
   504→        </form>
   505→      </DialogContent>
   506→    </Dialog>
   507→  );
   508→}

   514→function NewCampaignDialog({
   515→  open,
   516→  onOpenChange,
   517→  segments,
   518→  templates,
   519→  onCreate,
   520→}: {
   521→  open: boolean;
   522→  onOpenChange: (v: boolean) => void;
   523→  segments: Segment[];
   524→  templates: Template[];
   525→  onCreate: (c: Omit<Campaign, "id" | "status" | "sent" | "openRate" | "ctr" | "createdAt">) => void;
   526→}) {
   527→  const [name, setName] = React.useState("");
   528→  const [segmentId, setSegmentId] = React.useState<string>("");
   529→  const [channel, setChannel] = React.useState<Channel>("email");
   530→  const [templateId, setTemplateId] = React.useState<string>("");
   531→  const [errors, setErrors] = React.useState<{ name?: string; segmentId?: string; templateId?: string }>({});
   532→
   533→  React.useEffect(() => {
   534→    if (open) {
   535→      setName("");
   536→      setSegmentId(segments[0]?.id ?? "");
   537→      setChannel("email");
   538→      setTemplateId(templates[0]?.id ?? "");
   539→      setErrors({});
   540→    }
   541→  }, [open, segments, templates]);
   542→
   543→  function handleSubmit(ev: React.FormEvent) {
   544→    ev.preventDefault();
   545→    const e: typeof errors = {};
   546→    if (!name.trim()) e.name = "El nombre es obligatorio.";
   547→    if (!segmentId) e.segmentId = "Selecciona un segmento.";
   548→    if (!templateId) e.templateId = "Selecciona una plantilla.";
   549→    setErrors(e);
   550→    if (Object.keys(e).length > 0) return;
   551→    onCreate({ name: name.trim(), segmentId, channel, templateId });
   552→    onOpenChange(false);
   553→  }
   554→
   555→  return (
   556→    <Dialog open={open} onOpenChange={onOpenChange}>
   557→      <DialogContent className="sm:max-w-lg">
   558→        <DialogHeader>
   559→          <DialogTitle>Nueva campaña</DialogTitle>
   560→          <DialogDescription>
   561→            La campaña se crea como borrador. Podrás activarla cuando esté lista.
   562→          </DialogDescription>
   563→        </DialogHeader>
   564→        <form onSubmit={handleSubmit} className="space-y-4" noValidate>
   565→          <div className="space-y-2">
   566→            <label htmlFor="camp-name" className="text-xs font-mono uppercase tracking-wider text-muted-foreground">
   567→              Nombre <span className="text-destructive">*</span>
   568→            </label>
   569→            <Input
   570→              id="camp-name"
   571→              value={name}

  1143→  const [segments, setSegments] = React.useState<Segment[]>(DEMO_SEGMENTS);
  1144→  const [campaigns, setCampaigns] = React.useState<Campaign[]>(DEMO_CAMPAIGNS);
  1145→  const [templates, setTemplates] = React.useState<Template[]>(DEMO_TEMPLATES);
  1146→
  1147→  const [newSegmentOpen, setNewSegmentOpen] = React.useState(false);

  1150→
  1151→  const [selectedCampaignId, setSelectedCampaignId] = React.useState<string | null>(null);

  1153→  const selectedCampaign = campaigns.find((c) => c.id === selectedCampaignId) ?? null;
  1154→  const segmentOf = (id: string) => segments.find((s) => s.id === id);
  1155→  const templateOf = (id: string) => templates.find((t) => t.id === id);
  1156→
  1157→  function handleCreateSegment(s: Omit<Segment, "id" | "count">) {
  1158→    const newSeg: Segment = {
  1159→      ...s,
  1160→      id: `s${Date.now()}`,

  1165→
  1166→  function handleCreateCampaign(c: Omit<Campaign, "id" | "status" | "sent" | "openRate" | "ctr" | "createdAt">) {
  1167→    const today = new Date();
  1168→    const created = `${String(today.getDate()).padStart(2, "0")} ${today.toLocaleDateString("es-ES", { month: "short" })} ${today.getFullYear()}`;
  1169→    const newCamp: Campaign = {
  1170→      ...c,
  1171→      id: `c${Date.now()}`,
  1172→      status: "borrador",
  1173→      sent: 0,
  1174→      openRate: 0,
  1175→      ctr: 0,
  1176→      createdAt: created,
  1177→    };

  1179→  }
  1180→
  1181→  function handleCreateTemplate(t: Omit<Template, "id" | "lastUsed">) {

  1198→      sent: 0,
  1199→      openRate: 0,
