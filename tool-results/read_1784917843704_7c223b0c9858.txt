   248→function normalize(s: string): string {
   249→  return s
   250→    .toLowerCase()
   251→    .trim()
   252→    .replace(/\s+/g, " ")
   253→    .replace(/[¿?¡!.,;:]/g, "")
   254→    .trim();
   255→}
   256→
   257→function nowTime(): string {
   258→  return new Date().toLocaleTimeString("es-ES", { hour: "2-digit", minute: "2-digit" });
   259→}
   260→
   261→function hasPermission(role: Role, resp: DemoResp | undefined): boolean {
   262→  if (!resp?.restricted) return true;
   263→  return role === "owner" || role === "manager";
   264→}
   265→
   266→function confidenceTone(c: Confidence): { cls: string; label: string; pct: string } {
   267→  if (c === "alta") return { cls: "border-emerald-400/40 bg-emerald-400/10 text-emerald-300", label: "Confianza: Alta", pct: "92%" };
   268→  if (c === "media") return { cls: "border-amber-400/40 bg-amber-400/10 text-amber-300", label: "Confianza: Media", pct: "78%" };
   269→  return { cls: "border-rose-400/40 bg-rose-400/10 text-rose-300", label: "Confianza: Baja", pct: "45%" };
   270→}

   276→function FloatingButton({ open, onToggle, hasInsight }: { open: boolean; onToggle: () => void; hasInsight: boolean }) {
   277→  const reduce = useReducedMotion();
   278→  return (
   279→    <motion.button
   280→      onClick={onToggle}
   281→      aria-label={open ? "Cerrar Copilot IA" : "Abrir Copilot IA"}
   282→      aria-expanded={open}
   283→      initial={reduce ? false : { scale: 0, opacity: 0 }}
   284→      animate={{ scale: 1, opacity: 1 }}
   285→      transition={{ type: "spring", stiffness: 280, damping: 22, delay: 0.2 }}
   286→      whileHover={reduce ? undefined : { scale: 1.06 }}
   287→      whileTap={reduce ? undefined : { scale: 0.95 }}
   288→      className={cn(
   289→        "fixed bottom-4 right-4 sm:bottom-6 sm:right-6 z-50 h-14 w-14 sm:h-16 sm:w-16 rounded-full flex items-center justify-center text-black shadow-2xl",
   290→        "bg-gradient-to-br from-[var(--gold-soft)] via-[var(--gold)] to-[var(--gold-deep)]",
   291→        "ring-2 ring-[var(--gold)]/40 ring-offset-2 ring-offset-background"
   292→      )}
   293→    >
   294→      {/* Glow ring */}
   295→      <span
   296→        aria-hidden
   297→        className="absolute inset-0 rounded-full opacity-60 animate-ping"
   298→        style={{ background: "radial-gradient(circle, color-mix(in oklab, var(--gold) 35%, transparent) 0%, transparent 70%)" }}
   299→      />
   300→      <AnimatePresence mode="wait" initial={false}>
   301→        {open ? (
   302→          <motion.span
   303→            key="x"
   304→            initial={reduce ? false : { rotate: -90, opacity: 0 }}
   305→            animate={{ rotate: 0, opacity: 1 }}
   306→            exit={reduce ? undefined : { rotate: 90, opacity: 0 }}
   307→            transition={{ duration: 0.15 }}
   308→            className="relative"
   309→          >
   310→            <X className="h-6 w-6" />
   311→          </motion.span>
   312→        ) : (
   313→          <motion.span
   314→            key="sp"
   315→            initial={reduce ? false : { rotate: 90, opacity: 0 }}
   316→            animate={{ rotate: 0, opacity: 1 }}
   317→            exit={reduce ? undefined : { rotate: -90, opacity: 0 }}
   318→            transition={{ duration: 0.15 }}
   319→            className="relative"
   320→          >
   321→            <Sparkles className="h-6 w-6" />
   322→          </motion.span>
   323→        )}
   324→      </AnimatePresence>
   325→      {/* Notification dot */}
   326→      {hasInsight && !open && (
   327→        <span className="absolute -top-0.5 -right-0.5 h-3.5 w-3.5 rounded-full bg-[var(--teal)] ring-2 ring-background flex items-center justify-center">
   328→          <span className="h-1.5 w-1.5 rounded-full bg-black/60" />
   329→        </span>
   330→      )}
   331→    </motion.button>
   332→  );
   333→}

   358→
   359→function AiMessage({ msg }: { msg: Msg }) {
   360→  const reduce = useReducedMotion();
   361→  const go = useNav((s) => s.go);
   362→  const conf = msg.confidence ? confidenceTone(msg.confidence) : null;
   363→
   364→  return (
   365→    <motion.div
   366→      initial={reduce ? false : { opacity: 0, y: 6 }}
   367→      animate={{ opacity: 1, y: 0 }}
   368→      transition={{ duration: 0.25 }}
   369→      className="flex gap-2.5"
   370→    >
   371→      <div className="h-7 w-7 rounded-md bg-gradient-to-br from-[var(--gold)] to-[var(--gold-deep)] flex items-center justify-center text-black shrink-0">
   372→        <Sparkles className="h-3.5 w-3.5" />
   373→      </div>
   374→      <div className="min-w-0 flex-1 space-y-2">
   375→        <div className="rp-glass rounded-2xl rounded-tl-sm px-3.5 py-2.5 text-sm leading-relaxed text-foreground/90">
   376→          {msg.content}
   377→          {msg.streaming && <span className="inline-block w-1 h-3.5 ml-0.5 bg-[var(--gold-soft)] animate-pulse align-middle" aria-hidden />}
   378→        </div>
   379→
   380→        {/* Sources + confidence row */}
   381→        {!msg.streaming && !msg.denied && !msg.noData && (
   382→          <div className="flex flex-wrap items-center gap-1.5">
   383→            {msg.sources?.map((s) => (
   384→              <span
   385→                key={s.label}
   386→                className="inline-flex items-center gap-1 rounded-md border border-[var(--teal)]/30 bg-[var(--teal)]/[0.08] px-1.5 py-0.5 text-[10px] font-mono uppercase tracking-wider text-[var(--teal)]"
   387→              >
   388→                <span className="h-1 w-1 rounded-full bg-[var(--teal)]" />
   389→                {s.label}
   390→              </span>
   391→            ))}
   392→            {conf && (
   393→              <span className={cn("inline-flex items-center gap-1 rounded-md border px-1.5 py-0.5 text-[10px] font-mono uppercase tracking-wider", conf.cls)}>
   394→                {conf.label}
   395→              </span>
   396→            )}
   397→          </div>
   398→        )}
   399→
   400→        {/* Ver en módulo */}
   401→        {!msg.streaming && msg.viewIn && !msg.denied && !msg.noData && (
   402→          <button
   403→            onClick={() => go(msg.viewIn!)}
   404→            className="inline-flex items-center gap-1.5 rounded-md border border-[var(--gold)]/40 bg-[var(--gold)]/[0.08] px-2.5 py-1.5 text-[11px] font-medium text-[var(--gold-soft)] hover:bg-[var(--gold)]/[0.16] transition-colors min-h-[36px]"
   405→          >
   406→            <ArrowRight className="h-3 w-3" />
   407→            {msg.viewInLabel}
   408→            <ChevronRight className="h-3 w-3 opacity-60" />
   409→          </button>
   410→        )}
   411→
   412→        {/* No data notice */}
   413→        {!msg.streaming && msg.noData && (
   414→          <div className="flex items-center gap-1.5 text-[11px] text-amber-300">
   415→            <AlertTriangle className="h-3 w-3" />
   416→            <span>No tengo datos suficientes</span>
   417→          </div>
   418→        )}
   419→
   420→        {/* Denied notice */}
   421→        {!msg.streaming && msg.denied && (
   422→          <div className="flex items-center gap-1.5 text-[11px] text-rose-300">
   423→            <Lock className="h-3 w-3" />
   424→            <span>Permiso requerido</span>
   425→          </div>
   426→        )}
   427→      </div>
   428→    </motion.div>
   429→  );
   430→}

   456→function RoleSelector({ role, onChange }: { role: Role; onChange: (r: Role) => void }) {
   457→  const roles: { id: Role; label: string }[] = [
   458→    { id: "owner", label: "Owner" },
   459→    { id: "manager", label: "Manager" },
   460→    { id: "staff", label: "Staff" },
   461→  ];
   462→  return (
   463→    <div className="flex items-center gap-1 rounded-md border border-border/60 bg-foreground/[0.03] p-0.5">
   464→      {roles.map((r) => (
   465→        <button
   466→          key={r.id}
   467→          onClick={() => onChange(r.id)}
   468→          className={cn(
   469→            "h-6 rounded px-2 text-[10px] font-mono uppercase tracking-wider transition-colors",
   470→            role === r.id
   471→              ? "bg-[var(--gold)]/20 text-[var(--gold-soft)]"
   472→              : "text-muted-foreground hover:text-foreground"
   473→          )}
   474→          aria-pressed={role === r.id}
   475→        >
   476→          {r.label}
   477→        </button>
   478→      ))}
   479→    </div>
   480→  );
   481→}

   542→  function findResponse(query: string): DemoResp | undefined {
   543→    const norm = normalize(query);
   544→    // Exact match
   545→    if (DEMO_RESPONSES[norm]) return DEMO_RESPONSES[norm];
   546→    // Fuzzy: keyword match
   547→    for (const key of Object.keys(DEMO_RESPONSES)) {
   548→      const keyTokens = key.split(" ").filter((t) => t.length > 3);
   549→      const matches = keyTokens.filter((t) => norm.includes(t));
   550→      if (matches.length >= Math.ceil(keyTokens.length * 0.6)) {
   551→        return DEMO_RESPONSES[key];
   552→      }
   553→    }
   554→    return undefined;
   555→  }

   557→  function streamResponse(id: string, fullText: string) {
   558→    setStreamingId(id);
   559→    let i = 0;
   560→    const chunkSize = Math.max(2, Math.ceil(fullText.length / 50));
   561→    streamTimer.current = setInterval(() => {
   562→      i += chunkSize;
   563→      const partial = fullText.slice(0, i);
   564→      setMessages((prev) =>
   565→        prev.map((m) => (m.id === id ? { ...m, content: partial } : m))
   566→      );
   567→      if (i >= fullText.length) {
   568→        if (streamTimer.current) clearInterval(streamTimer.current);
   569→        streamTimer.current = null;
   570→        setStreamingId(null);
   571→        setMessages((prev) =>
   572→          prev.map((m) => (m.id === id ? { ...m, streaming: false } : m))
   573→        );
   574→      }
   575→    }, 20);
   576→  }

   578→  function send(text: string) {
   579→    const trimmed = text.trim();
   580→    if (!trimmed || isTyping || streamingId) return;
   581→
   582→    const userMsg: Msg = {
   583→      id: `u-${Date.now()}`,
   584→      role: "user",
   585→      content: trimmed,
   586→      ts: nowTime(),
   587→    };
   588→    setMessages((prev) => [...prev, userMsg]);
   589→    setInput("");
   590→    setIsTyping(true);
   591→
   592→    // Find response
   593→    const resp = findResponse(trimmed);
   594→
   595→    // Simulate thinking delay then stream
   596→    setTimeout(() => {
   597→      setIsTyping(false);
   598→
   599→      // Permission check
   600→      if (resp && !hasPermission(role, resp)) {
   601→        const deniedMsg: Msg = {
   602→          id: `a-${Date.now()}`,
   603→          role: "ai",
   604→          content: "Tu rol no permite consultar esta información. Esta consulta requiere permiso de Manager o superior. Contacta con un administrador de la organización si necesitas acceso.",
   605→          ts: nowTime(),
   606→          denied: true,
   607→        };
   608→        setMessages((prev) => [...prev, deniedMsg]);
   609→        return;
   610→      }
   611→
   612→      if (!resp) {
   613→        // No data
   614→        const noDataMsg: Msg = {
   615→          id: `a-${Date.now()}`,
   616→          role: "ai",
   617→          content: "No tengo datos suficientes para responder a esa consulta. (demo) Prueba con una de las acciones sugeridas o reformula tu pregunta sobre reservas, clientes, ocupación, campañas o facturación.",
   618→          ts: nowTime(),
   619→          noData: true,
   620→        };
   621→        setMessages((prev) => [...prev, noDataMsg]);
   622→        return;
   623→      }
   624→
   625→      const aiMsg: Msg = {
   626→        id: `a-${Date.now()}`,
   627→        role: "ai",
   628→        content: "",
   629→        ts: nowTime(),
   630→        streaming: true,
   631→        sources: resp.sources.map((s) => ({ label: s })),
   632→        confidence: resp.confidence,
   633→        viewIn: resp.viewIn?.section,
   634→        viewInLabel: resp.viewIn?.label,
   635→      };
   636→      setMessages((prev) => [...prev, aiMsg]);
   637→      // Begin streaming
   638→      streamResponse(aiMsg.id, resp.text);
   639→    }, 650);
   640→  }

   642→  function onKeyDown(e: React.KeyboardEvent<HTMLInputElement>) {
   643→    if (e.key === "Enter" && !e.shiftKey) {
   644→      e.preventDefault();
   645→      send(input);
   646→    }
   647→  }
   648→
   649→  return (

   651→      initial={reduce ? false : { opacity: 0, y: 24, scale: 0.96 }}

   654→      transition={{ type: "spring", stiffness: 320, damping: 30 }}
   655→      role="dialog"
   656→      aria-label="Copilot IA"
   657→      className={cn(
   658→        "fixed z-50 flex flex-col rp-glass-strong rounded-t-2xl sm:rounded-2xl overflow-hidden",
   659→        "left-0 right-0 bottom-0 sm:left-auto sm:right-4 sm:bottom-24 sm:w-[400px]",
   660→        "h-[min(560px,85vh)] sm:h-[min(560px,calc(100vh-8rem))]",
   661→        "ring-1 ring-[var(--gold)]/30 shadow-2xl"
   662→      )}
   663→    >
   664→      {/* Header */}
   665→      <header className="flex items-center gap-2 px-3 sm:px-4 h-14 border-b border-border/60 bg-gradient-to-r from-[var(--gold)]/[0.08] to-transparent shrink-0">
   666→        <div className="h-8 w-8 rounded-md bg-gradient-to-br from-[var(--gold)] to-[var(--gold-deep)] flex items-center justify-center text-black shrink-0">
   667→          <Sparkles className="h-4 w-4" />
   668→        </div>
   669→        <div className="flex-1 min-w-0">
   670→          <div className="flex items-center gap-2">
   671→            <span className="font-display text-sm font-medium truncate">Copilot IA</span>
   672→            <span className="inline-flex items-center gap-1 rounded border border-[var(--gold)]/30 bg-[var(--gold)]/[0.08] px-1.5 py-0.5 text-[9px] font-mono uppercase tracking-wider text-[var(--gold-soft)] shrink-0">
   673→              <Cpu className="h-2.5 w-2.5" />
   674→              Llama 3.1 8B
   675→            </span>
   676→          </div>

   678→            <span className="relative flex h-1.5 w-1.5">

   800→export function AiCopilot() {
   801→  const [open, setOpen] = React.useState(false);
   802→  const [minimized, setMinimized] = React.useState(false);
   803→  const [role, setRole] = React.useState<Role>("owner");
   804→  const [hasInsight] = React.useState(true);
   805→
   806→  // Open panel (un-minimizes if was minimized)
   807→  function handleToggle() {
   808→    if (open && !minimized) {
   809→      setOpen(false);
   810→    } else {
   811→      setOpen(true);
   812→      setMinimized(false);
   813→    }
   814→  }
   815→
   816→  function handleClose() {
   817→    setOpen(false);
   818→    setMinimized(false);
   819→  }
   820→
   821→  function handleMinimize() {
   822→    setMinimized(true);
   823→  }
   824→
   825→  return (
   826→    <>
   827→      <FloatingButton open={open && !minimized} onToggle={handleToggle} hasInsight={hasInsight} />
   828→      <AnimatePresence>
   829→        {open && !minimized && (
   830→          <CopilotPanel
   831→            onClose={handleClose}
   832→            onMinimize={handleMinimize}
   833→            role={role}
   834→            onRoleChange={setRole}
   835→          />
   836→        )}
   837→      </AnimatePresence>
   838→    </>
   839→  );
   840→}
