   518→function IconBtn({
   519→  icon: Icon,
   520→  label,
   521→  onClick,
   522→  tone = "default",
   523→}: {
   524→  icon: React.ElementType;
   525→  label: string;
   526→  onClick?: () => void;
   527→  tone?: "default" | "amber" | "destructive" | "teal";
   528→}) {
   529→  const cls =
   530→    tone === "amber"
   531→      ? "text-amber-300 hover:bg-amber-400/10 border-amber-400/30"
   532→      : tone === "destructive"
   533→      ? "text-rose-300 hover:bg-rose-500/10 border-rose-500/30"
   534→      : tone === "teal"
   535→      ? "text-[var(--teal)] hover:bg-[var(--teal)]/10 border-[var(--teal)]/30"
   536→      : "text-muted-foreground hover:bg-foreground/5 border-border/40";
   537→  return (
   538→    <TooltipProvider delayDuration={150}>
   539→      <Tooltip>
   540→        <TooltipTrigger asChild>
   541→          <button
   542→            onClick={onClick}
   543→            aria-label={label}
   544→            className={cn("inline-flex h-9 w-9 items-center justify-center rounded-md border transition-colors", cls)}
   545→          >
   546→            <Icon className="h-3.5 w-3.5" />
