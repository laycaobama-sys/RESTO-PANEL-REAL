   385→/* =========================================================
   386→ * Helpers
   387→ * =======================================================*/
   388→function loadColor(score: number) {
   389→  if (score >= 75) return { text: "text-destructive", bg: "bg-destructive", ring: "ring-destructive/40" };
   390→  if (score >= 50) return { text: "rp-gold-text", bg: "bg-[var(--gold)]", ring: "ring-[var(--gold)]/40" };
   391→  return { text: "text-emerald-300", bg: "bg-emerald-400", ring: "ring-emerald-400/40" };
   392→}
   393→
   394→function loadLabel(score: number) {
   395→  if (score >= 75) return "Alta";
   396→  if (score >= 50) return "Media";
   397→  return "Baja";
   398→}
   399→
   400→function initials(name: string) {
   401→  const parts = name.split(" ");
   402→  return (parts[0]?.[0] ?? "") + (parts[1]?.[0] ?? "");
   403→}
   404→
   405→function avatarGradient(name: string) {
   406→  const grads = [
   407→    "from-amber-400/80 to-[var(--gold-deep)]",
   408→    "from-[var(--teal)]/80 to-[var(--teal-deep)]",
   409→    "from-purple-400/80 to-purple-700",
   410→    "from-rose-400/80 to-rose-700",
   411→    "from-sky-400/80 to-sky-700",
   412→    "from-emerald-400/80 to-emerald-700",
   413→  ];
   414→  const idx = name.charCodeAt(0) % grads.length;
   415→  return grads[idx];
   416→}

   418→/* =========================================================
   419→ * Circular load meter
   420→ * =======================================================*/
   421→function LoadMeter({
   422→  score,
   423→  size = 92,
   424→  reduce,
   425→}: {
   426→  score: number;
   427→  size?: number;
   428→  reduce: boolean;
   429→}) {
   430→  const stroke = 7;
   431→  const r = (size - stroke * 2) / 2;
   432→  const c = 2 * Math.PI * r;
   433→  const offset = c - (score / 100) * c;
   434→  const lc = loadColor(score);
   435→
   436→  return (
   437→    <div className="relative grid place-items-center" style={{ width: size, height: size }}>
   438→      <svg width={size} height={size} className="-rotate-90">
   439→        <circle
   440→          cx={size / 2}
   441→          cy={size / 2}
   442→          r={r}
   443→          fill="none"
   444→          stroke="currentColor"
   445→          strokeWidth={stroke}
   446→          className="text-foreground/10"
   447→        />
   448→        <motion.circle
   449→          cx={size / 2}
   450→          cy={size / 2}
   451→          r={r}
   452→          fill="none"
   453→          stroke="currentColor"
   454→          strokeWidth={stroke}
   455→          strokeLinecap="round"
   456→          strokeDasharray={c}
   457→          initial={reduce ? false : { strokeDashoffset: c }}
   458→          animate={{ strokeDashoffset: offset }}
   459→          transition={{ duration: 0.8, ease: "easeOut" }}
   460→          className={lc.text}
   461→        />
   462→      </svg>
   463→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   464→        <span className={cn("font-display text-2xl font-light tabular-nums", lc.text)}>
   465→          {score}
   466→          <span className="text-xs text-muted-foreground">%</span>
   467→        </span>
   468→        <span className="mt-0.5 text-[9px] font-mono uppercase tracking-wider text-muted-foreground">
   469→          {loadLabel(score)}
   470→        </span>
   471→      </div>
   472→    </div>
   473→  );
   474→}
