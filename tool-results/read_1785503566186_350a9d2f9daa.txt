    48→/* ============================================================
    49→   Hook: prefers-reduced-motion
    50→============================================================ */
    51→function usePrefersReducedMotion() {
    52→  const [reduce, setReduce] = React.useState(false);
    53→  React.useEffect(() => {
    54→    const mq = window.matchMedia("(prefers-reduced-motion: reduce)");
    55→    setReduce(mq.matches);
    56→    const onChange = () => setReduce(mq.matches);
    57→    mq.addEventListener("change", onChange);
    58→    return () => mq.removeEventListener("change", onChange);
    59→  }, []);
    60→  return reduce;
    61→}

    63→/* ============================================================
    64→   Hook: useLocalStorage
    65→============================================================ */
    66→function useLocalStorage<T>(key: string, initial: T): [T, (v: T | ((p: T) => T)) => void] {
    67→  const [value, setValue] = React.useState<T>(() => {
    68→    if (typeof window === "undefined") return initial;
    69→    try {
    70→      const stored = window.localStorage.getItem(key);
    71→      return stored ? (JSON.parse(stored) as T) : initial;
    72→    } catch {
    73→      return initial;
    74→    }
    75→  });
    76→  React.useEffect(() => {
    77→    try {
    78→      window.localStorage.setItem(key, JSON.stringify(value));
    79→    } catch {
    80→      /* noop */
    81→    }
    82→  }, [key, value]);
    83→  return [value, setValue];
    84→}

   449→function UserCard({ expanded }: { expanded: boolean }) {

   452→  const openAuth = useNav((s) => s.openAuth);

   457→  // Close on outside click
   458→  React.useEffect(() => {
   459→    if (!menuOpen) return;
   460→    function onDown(e: MouseEvent) {
   461→      if (menuRef.current && !menuRef.current.contains(e.target as Node)) {
   462→        setMenuOpen(false);
   463→      }

   466→      if (e.key === "Escape") setMenuOpen(false);
   467→    }

   469→    document.addEventListener("keydown", onKey);
   470→    return () => {
   471→      document.removeEventListener("mousedown", onDown);
   472→      document.removeEventListener("keydown", onKey);
   473→    };
   474→  }, [menuOpen]);
   475→
   476→  const initials = user?.initials ?? "AM";
   477→  const name = user?.name ?? "Ana Martínez";
   478→  const role = user?.role ?? "Owner · Ramses Group";
   479→
   480→  return (
   481→    <div className="relative border-t border-border/60 p-2" ref={menuRef}>
   482→      <button
   483→        onClick={() => setMenuOpen((v) => !v)}
   484→        className={cn(
   485→          "w-full flex items-center rounded-md hover:bg-foreground/5 transition-colors min-h-[44px]",
   486→          expanded ? "gap-2.5 px-2 py-1.5" : "justify-center px-0 py-2",
   487→        )}

   494→          {initials}
   495→        </div>
   496→        {expanded && (
   497→          <div className="flex-1 min-w-0 text-left rp-fade-in">
   498→            <div className="text-sm font-medium truncate">{name}</div>
   499→            <div className="text-[11px] text-muted-foreground truncate">{role}</div>
   500→          </div>
   501→        )}
   502→        {expanded && <ChevronDown className="h-4 w-4 text-muted-foreground shrink-0" aria-hidden />}
   503→      </button>
   504→
   505→      {menuOpen && (

  1094→function PasswordInput({
  1095→  label,
  1096→  value,
  1097→  onChange,
  1098→  error,
  1099→  autoComplete,
  1100→  showStrength,
  1101→}: {
  1102→  label: string;
  1103→  value: string;
  1104→  onChange: (v: string) => void;
  1105→  error?: string | null;
  1106→  autoComplete?: string;
  1107→  showStrength?: boolean;
  1108→}) {
  1109→  const [show, setShow] = React.useState(false);
  1110→  const strength = React.useMemo(() => {
  1111→    if (!showStrength) return null;
  1112→    let score = 0;
  1113→    if (value.length >= 8) score++;
  1114→    if (/[A-Z]/.test(value)) score++;
  1115→    if (/[a-z]/.test(value)) score++;
  1116→    if (/[0-9]/.test(value)) score++;
  1117→    if (/[^A-Za-z0-9]/.test(value)) score++;
  1118→    return score;
  1119→  }, [value, showStrength]);
  1120→
  1121→  const strengthLabel = strength === null ? null : strength <= 1 ? "Débil" : strength <= 3 ? "Media" : "Fuerte";
  1122→  const strengthCls = strength === null ? "" : strength <= 1 ? "bg-rose-500" : strength <= 3 ? "bg-amber-400" : "bg-[var(--teal)]";
  1123→
  1124→  return (
  1125→    <div>
  1126→      <Label className="mb-1.5 block text-xs">{label}</Label>
  1127→      <div className="relative">
  1128→        <Lock className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground pointer-events-none" aria-hidden />
  1129→        <input
  1130→          type={show ? "text" : "password"}
  1131→          value={value}
  1132→          onChange={(e) => onChange(e.target.value)}
  1133→          autoComplete={autoComplete}
  1134→          className={cn(
  1135→            "w-full h-11 rounded-md border bg-background/40 pl-10 pr-10 text-sm outline-none transition-colors",
  1136→            error ? "border-rose-400/60 focus:border-rose-400" : "border-border/60 focus:border-[var(--gold)]/50",
  1137→          )}
  1138→          aria-invalid={!!error}
  1139→        />
  1140→        <button
  1141→          type="button"
  1142→          onClick={() => setShow((v) => !v)}
  1143→          className="absolute right-2 top-1/2 -translate-y-1/2 h-8 w-8 flex items-center justify-center rounded text-muted-foreground hover:text-foreground transition-colors"
  1144→          aria-label={show ? "Ocultar contraseña" : "Mostrar contraseña"}
  1145→          tabIndex={-1}
  1146→        >
  1147→          {show ? <EyeOff className="h-4 w-4" /> : <Eye className="h-4 w-4" />}
  1148→        </button>
  1149→      </div>
  1150→      {showStrength && strength !== null && value.length > 0 && (
  1151→        <div className="mt-1.5 flex items-center gap-2">
  1152→          <div className="flex-1 h-1 rounded-full bg-foreground/10 overflow-hidden">
  1153→            <div
  1154→              className={cn("h-full rounded-full transition-all", strengthCls)}
  1155→              style={{ width: `${(strength / 5) * 100}%` }}
  1156→            />
  1157→          </div>
  1158→          <span className="text-[10px] font-mono text-muted-foreground">{strengthLabel}</span>
  1159→        </div>
  1160→      )}
  1161→      <FieldError msg={error} />
  1162→    </div>
  1163→  );
  1164→}
