   972→function BuilderTab({ reduce }: { reduce: boolean | null }) {
   973→  const [nodes, setNodes] = React.useState<FlowNode[]>(FLOW_NODES);
   974→  const [selectedId, setSelectedId] = React.useState<string>("n1");
   975→  const [testOpen, setTestOpen] = React.useState(false);

   980→
   981→  const selected = nodes.find((n) => n.id === selectedId);
   982→
   983→  function addNode(type: NodeType, subtype: string, title: string) {
   984→    const id = `n${Date.now()}`;
   985→    setNodes((prev) => [...prev, { id, type, subtype, title, summary: "Configurar…" }]);
   986→    setSelectedId(id);
   987→    toast({ title: "Nodo añadido", description: title });

   990→  function deleteNode(id: string) {
   991→    setNodes((prev) => prev.filter((n) => n.id !== id));

   995→
