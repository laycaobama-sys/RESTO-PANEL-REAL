   365→function initials(name: string): string {
   366→  return name
   367→    .split(" ")
   368→    .slice(0, 2)
   369→    .map((p) => p[0])
   370→    .join("")
   371→    .toUpperCase();
   372→}
   373→
   374→function formatEur(n: number): string {
   375→  return n.toLocaleString("es-ES", {
   376→    style: "currency",
   377→    currency: "EUR",
   378→    maximumFractionDigits: 0,
   379→  });
   380→}
   381→
   382→function formatDate(iso: string): string {
   383→  // iso "2025-03-12" → "12 mar 2025"
   384→  const d = new Date(iso + "T00:00:00");
   385→  return d.toLocaleDateString("es-ES", {
   386→    day: "2-digit",
   387→    month: "short",
   388→    year: "numeric",
   389→  });
   390→}

   827→/* ------------------------------------------------------------------ */
   828→
   829→function Stars({ value, size = "sm" }: { value: number; size?: "sm" | "md" }) {
   830→  const dim = size === "md" ? "h-4 w-4" : "h-3 w-3";
   831→  const full = Math.floor(value);
   832→  const half = value - full >= 0.5;
   833→  return (
   834→    <span
   835→      className="inline-flex items-center gap-0.5"
   836→      role="img"
   837→      aria-label={`${value.toFixed(1)} de 5 estrellas`}
   838→    >
   839→      {Array.from({ length: 5 }).map((_, i) => {
   840→        const filled = i < full;
   841→        const isHalf = !filled && half && i === full;
   842→        return (
   843→          <Star
   844→            key={i}
   845→            className={cn(
   846→              dim,
   847→              filled || isHalf
   848→                ? "fill-[var(--gold)] text-[var(--gold)]"
   849→                : "fill-transparent text-muted-foreground/40"
   850→            )}
   851→            aria-hidden
   852→          />
   853→        );
   854→      })}
   855→    </span>
   856→  );
   857→}
