   282→function confidenceClass(c: number) {
   283→  if (c >= 85) return "bg-emerald-400/10 text-emerald-300 border-emerald-400/30";
   284→  if (c >= 65) return "bg-amber-400/10 text-amber-300 border-amber-400/30";
   285→  return "bg-destructive/10 text-destructive border-destructive/30";
   286→}
   287→
   288→function scoreColor(score: number) {
   289→  if (score >= 85) return "text-emerald-300";
   290→  if (score >= 70) return "rp-gold-text";
   291→  if (score >= 50) return "text-amber-300";
   292→  return "text-destructive";
   293→}

   440→/* =========================================================
   441→ * Reason row — single factor with impact bar
   442→ * =======================================================*/
   443→function ReasonRow({
   444→  reason,
   445→  index,
   446→  reduce,
   447→}: {
   448→  reason: { factor: string; impact: number; detail: string };
   449→  index: number;
   450→  reduce: boolean;
   451→}) {
   452→  const positive = reason.impact >= 0;
   453→  const maxAbs = 25;
   454→  const width = Math.min(Math.abs(reason.impact) / maxAbs, 1) * 100;
   455→
   456→  return (
   457→    <motion.li
   458→      initial={reduce ? false : { opacity: 0, x: -8 }}
   459→      animate={{ opacity: 1, x: 0 }}
   460→      transition={{ duration: 0.25, delay: reduce ? 0 : index * 0.04 }}
   461→      className="rp-glass rounded-xl p-3 sm:p-3.5"
   462→    >
   463→      <div className="flex items-start justify-between gap-3">
   464→        <div className="min-w-0">
   465→          <div className="flex items-center gap-2">
   466→            <span className="text-sm font-medium text-foreground">{reason.factor}</span>
   467→            <span
   468→              className={cn(
   469→                "inline-flex items-center gap-0.5 rounded-md border px-1.5 py-0.5 text-[10px] font-mono tabular-nums",
   470→                positive
   471→                  ? "border-emerald-400/30 bg-emerald-400/10 text-emerald-300"
   472→                  : "border-destructive/30 bg-destructive/10 text-destructive"
   473→              )}
   474→            >
   475→              {positive ? "+" : ""}
   476→              {reason.impact}
   477→            </span>
   478→          </div>
   479→          <p className="mt-1 text-xs leading-relaxed text-muted-foreground">
   480→            {reason.detail}
   481→          </p>
   482→        </div>
   483→      </div>
   484→      {/* Impact bar */}
   485→      <div className="mt-2.5 h-1 w-full overflow-hidden rounded-full bg-foreground/[0.08]">
   486→        <motion.div
   487→          initial={reduce ? false : { width: 0 }}
   488→          animate={{ width: `${width}%` }}
   489→          transition={{ duration: 0.5, delay: reduce ? 0 : 0.1 + index * 0.04, ease: "easeOut" }}
   490→          className={cn(
   491→            "h-full rounded-full",
   492→            positive
   493→              ? "bg-gradient-to-r from-emerald-400/60 to-emerald-400"
   494→              : "bg-gradient-to-r from-destructive/60 to-destructive"
   495→          )}
   496→        />
   497→      </div>
   498→    </motion.li>
   499→  );
   500→}
