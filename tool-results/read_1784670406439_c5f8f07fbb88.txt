   487→/* ---------- Sparkline ---------- */
   488→
   489→function Sparkline({ data, color }: { data: number[]; color: "gold" | "teal" }) {
   490→  const rawId = React.useId();
   491→  const id = rawId.replace(/[:]/g, "");
   492→  const w = 76;
   493→  const h = 28;
   494→  const pad = 3;
   495→  const max = Math.max(...data);
   496→  const min = Math.min(...data);
   497→  const range = max - min || 1;
   498→  const pts = data.map((v, i) => {
   499→    const x = (i / (data.length - 1)) * (w - pad * 2) + pad;
   500→    const y = h - pad - ((v - min) / range) * (h - pad * 2);
   501→    return [x, y] as [number, number];
   502→  });
   503→  const linePts = pts.map(([x, y]) => `${x.toFixed(1)},${y.toFixed(1)}`).join(" ");
   504→  const areaPath = `M${pts[0][0].toFixed(1)},${(h - pad).toFixed(1)} L${pts
   505→    .map(([x, y]) => `${x.toFixed(1)},${y.toFixed(1)}`)
   506→    .join(" L")} L${pts[pts.length - 1][0].toFixed(1)},${(h - pad).toFixed(1)} Z`;
   507→  const stroke = color === "gold" ? "var(--gold)" : "var(--teal)";
   508→  const lastPt = pts[pts.length - 1];
   509→
   510→  return (
   511→    <svg
   512→      width={w}
   513→      height={h}
   514→      viewBox={`0 0 ${w} ${h}`}
   515→      aria-hidden="true"
   516→      focusable="false"
   517→      className="shrink-0 overflow-visible"
   518→    >
   519→      <defs>
   520→        <linearGradient id={`spark-${id}`} x1="0" x2="0" y1="0" y2="1">
   521→          <stop offset="0%" stopColor={stroke} stopOpacity="0.32" />
   522→          <stop offset="100%" stopColor={stroke} stopOpacity="0" />
   523→        </linearGradient>
   524→      </defs>
   525→      <path d={areaPath} fill={`url(#spark-${id})`} />
   526→      <polyline
   527→        points={linePts}
   528→        fill="none"
   529→        stroke={stroke}
   530→        strokeWidth="1.5"
   531→        strokeLinejoin="round"
   532→        strokeLinecap="round"
   533→      />
   534→      <circle cx={lastPt[0]} cy={lastPt[1]} r="2" fill={stroke} />
   535→    </svg>
   536→  );
   537→}

   541→function KpiCard({ kpi }: { kpi: Kpi }) {
   542→  const Icon = kpi.icon;
   543→  const isGood = kpi.trend === "up" || kpi.goodDown === true;
   544→  const valueColor =
   545→    kpi.color === "gold" ? "text-[var(--gold-soft)]" : "text-[var(--teal)]";
   546→  const pillCls = isGood
   547→    ? kpi.color === "gold"
   548→      ? "bg-[var(--gold)]/10 text-[var(--gold-soft)] border-[var(--gold)]/25"
