   124→/* ---------------- MRR chart ---------------- */
   125→function MrrChart() {
   126→  const months = ["Ene", "Feb", "Mar", "Abr", "May", "Jun", "Jul", "Ago", "Sep", "Oct", "Nov", "Dic"];
   127→  const values = [28500, 30200, 32800, 35100, 37400, 39200, 41800, 43200, 44600, 45900, 47100, 48250];
   128→  const max = Math.max(...values);
   129→  const min = Math.min(...values);
   130→  const w = 720, h = 220, pad = 32;
   131→  const x = (i: number) => pad + (i * (w - pad * 2)) / (values.length - 1);
   132→  const y = (v: number) => h - pad - ((v - min) / (max - min)) * (h - pad * 2);
   133→  const pathLine = values.map((v, i) => `${i === 0 ? "M" : "L"}${x(i)},${y(v)}`).join(" ");
   134→  const pathArea = `${pathLine} L${x(values.length - 1)},${h - pad} L${x(0)},${h - pad} Z`;
   135→  return (
   136→    <svg viewBox={`0 0 ${w} ${h}`} className="w-full h-auto" role="img" aria-label="Crecimiento MRR últimos 12 meses">
   137→      <defs>
   138→        <linearGradient id="mrr-fill" x1="0" x2="0" y1="0" y2="1">
   139→          <stop offset="0%" stopColor="var(--gold)" stopOpacity="0.35" />
   140→          <stop offset="100%" stopColor="var(--gold)" stopOpacity="0" />
   141→        </linearGradient>
   142→        <linearGradient id="mrr-line" x1="0" x2="1" y1="0" y2="0">
   143→          <stop offset="0%" stopColor="var(--gold-soft)" />
   144→          <stop offset="100%" stopColor="var(--teal)" />
   145→        </linearGradient>
   146→      </defs>
   147→      {[0, 0.25, 0.5, 0.75, 1].map((p) => (
   148→        <line key={p} x1={pad} x2={w - pad} y1={pad + p * (h - pad * 2)} y2={pad + p * (h - pad * 2)} stroke="currentColor" strokeOpacity="0.06" />
   149→      ))}
   150→      <path d={pathArea} fill="url(#mrr-fill)" />
   151→      <path d={pathLine} fill="none" stroke="url(#mrr-line)" strokeWidth="2.5" strokeLinejoin="round" strokeLinecap="round" />
   152→      {values.map((v, i) => (
   153→        <g key={i}>
   154→          <circle cx={x(i)} cy={y(v)} r="3" fill="var(--background)" stroke="var(--gold)" strokeWidth="2" />
   155→          <text x={x(i)} y={h - pad + 16} fontSize="10" fill="currentColor" fillOpacity="0.5" textAnchor="middle" fontFamily="var(--font-jetbrains)">
   156→            {months[i]}
   157→          </text>
   158→        </g>
   159→      ))}
   160→    </svg>
   161→  );
   162→}
