   296→/* =========================================================
   297→ * Command palette (⌘K)
   298→ * Quick actions modal with search and keyboard navigation.
   299→ * Navigates via useNav.getState().go(section).
   300→ * =======================================================*/
   301→interface CmdAction {
   302→  id: string;
   303→  label: string;
   304→  hint: string;
   305→  icon: React.ElementType;
   306→  group: "Navegación" | "Acciones";
   307→  run: () => void;
   308→}
   309→
   310→function CommandPalette({ open, onOpenChange }: { open: boolean; onOpenChange: (o: boolean) => void }) {
   311→  const [query, setQuery] = React.useState("");
   312→  const [active, setActive] = React.useState(0);
   313→  const inputRef = React.useRef<HTMLInputElement>(null);
   314→  const listRef = React.useRef<HTMLDivElement>(null);
   315→
   316→  const actions = React.useMemo<CmdAction[]>(() => {
   317→    const nav: CmdAction[] = NAV.map((n) => ({
   318→      id: `nav-${n.id}`,
   319→      label: n.label,
   320→      hint: "Ir a sección",
   321→      icon: n.icon,
   322→      group: "Navegación",
   323→      run: () => useNav.getState().go(n.id),
   324→    }));
   325→    const quick: CmdAction[] = [
   326→      { id: "qa-executive", label: "Abrir Centro Ejecutivo", hint: "IA + Alertas", icon: Crown, group: "Acciones", run: () => useNav.getState().go("executive") },
   327→      { id: "qa-new-reserva", label: "Nueva reserva", hint: "Crear reserva", icon: CalendarPlus, group: "Acciones", run: () => useNav.getState().go("reservas") },
   328→      { id: "qa-search-client", label: "Buscar cliente", hint: "Abrir CRM", icon: UserSearch, group: "Acciones", run: () => useNav.getState().go("crm") },
   329→      { id: "qa-floor-plan", label: "Ver plano de mesas", hint: "Abrir reservas", icon: MapIcon, group: "Acciones", run: () => useNav.getState().go("reservas") },
   330→      { id: "qa-marketing", label: "Lanzar campaña", hint: "Marketing", icon: Megaphone, group: "Acciones", run: () => useNav.getState().go("marketing") },
   331→      { id: "qa-growth-analytics", label: "Ver Growth Analytics", hint: "ROI e insights IA", icon: TrendingUp, group: "Acciones", run: () => useNav.getState().go("growth-analytics") },
   332→      { id: "qa-growth-reputation", label: "Gestionar reputación", hint: "Reseñas + NPS", icon: Award, group: "Acciones", run: () => useNav.getState().go("growth-reputation") },
   333→      { id: "qa-reviews", label: "Responder reviews", hint: "Google Reviews", icon: Star, group: "Acciones", run: () => useNav.getState().go("reviews") },
   334→      { id: "qa-analytics", label: "Ver analítica", hint: "Analytics", icon: BarChart3, group: "Acciones", run: () => useNav.getState().go("analytics") },
   335→      { id: "qa-settings", label: "Configuración", hint: "Ajustes del local", icon: Settings, group: "Acciones", run: () => useNav.getState().go("configuracion") },
   336→    ];
   337→    return [...quick, ...nav];
   338→  }, []);
   339→
   340→  const filtered = React.useMemo(() => {
   341→    const q = query.trim().toLowerCase();
   342→    if (!q) return actions;
   343→    return actions.filter((a) =>
   344→      a.label.toLowerCase().includes(q) || a.hint.toLowerCase().includes(q)
   345→    );
   346→  }, [actions, query]);
   347→
   348→  // Reset state on open
   349→  React.useEffect(() => {
   350→    if (open) {
   351→      setQuery("");
   352→      setActive(0);
   353→      // focus input after dialog mounts
   354→      requestAnimationFrame(() => inputRef.current?.focus());
   355→    }
   356→  }, [open]);
   357→
   358→  // Keep active index in range
   359→  React.useEffect(() => {
   360→    if (active >= filtered.length) setActive(0);
   361→  }, [filtered, active]);
   362→
   363→  // Scroll active item into view
   364→  React.useEffect(() => {
   365→    const el = listRef.current?.querySelector<HTMLElement>(`[data-idx="${active}"]`);
   366→    el?.scrollIntoView({ block: "nearest" });
   367→  }, [active]);
   368→
   369→  function execute(a: CmdAction) {
   370→    a.run();
   371→    onOpenChange(false);
   372→  }
   373→
   374→  function onKeyDown(e: React.KeyboardEvent) {
   375→    if (e.key === "ArrowDown") {
   376→      e.preventDefault();
   377→      setActive((i) => (i + 1) % Math.max(filtered.length, 1));
   378→    } else if (e.key === "ArrowUp") {
   379→      e.preventDefault();
   380→      setActive((i) => (i - 1 + filtered.length) % Math.max(filtered.length, 1));
   381→    } else if (e.key === "Enter") {
   382→      e.preventDefault();
   383→      const a = filtered[active];
   384→      if (a) execute(a);
   385→    } else if (e.key === "Escape") {
   386→      onOpenChange(false);
   387→    }
   388→  }
   389→
   390→  // Group filtered actions for rendering
   391→  const groups = React.useMemo(() => {
   392→    const map = new Map<string, CmdAction[]>();
   393→    for (const a of filtered) {
   394→      const arr = map.get(a.group) ?? [];
   395→      arr.push(a);
   396→      map.set(a.group, arr);
   397→    }
   398→    return Array.from(map.entries());
   399→  }, [filtered]);
   400→
   401→  // Flat index lookup for keyboard nav
   402→  let flatIdx = -1;
   403→
   404→  return (
   405→    <Dialog open={open} onOpenChange={onOpenChange}>
   406→      <DialogContent
   407→        showCloseButton={false}
   408→        className="p-0 gap-0 sm:max-w-lg max-h-[80vh] overflow-hidden flex flex-col"
   409→        onKeyDown={onKeyDown}
   410→      >
   411→        <DialogTitle className="sr-only">Paleta de comandos</DialogTitle>
   412→        <DialogDescription className="sr-only">
   413→          Busca acciones rápidas o secciones y navega con el teclado.
   414→        </DialogDescription>
   415→        {/* Search input */}
   416→        <div className="flex items-center gap-2.5 px-4 h-14 border-b border-border/60">
   417→          <Search className="h-4 w-4 text-muted-foreground shrink-0" aria-hidden />
   418→          <input
   419→            ref={inputRef}

   529→class ErrorBoundary extends React.Component<
   530→  { children: React.ReactNode },
   531→  { hasError: boolean; error: Error | null }
   532→> {
   533→  constructor(props: { children: React.ReactNode }) {
   534→    super(props);
   535→    this.state = { hasError: false, error: null };
   536→  }
   537→  static getDerivedStateFromError(error: Error) {
   538→    return { hasError: true, error };
   539→  }
   540→  componentDidCatch(error: Error) {
   541→    console.error("SectionRenderer error:", error.message);
   542→  }
   543→  render() {
   544→    if (this.state.hasError) {
   545→      return (
   546→        <div className="rp-glass rounded-xl p-8 text-center max-w-2xl mx-auto">
   547→          <p className="text-sm text-destructive mb-3 font-medium">Error al cargar esta sección</p>
   548→          <p className="text-xs text-muted-foreground mb-4 font-mono break-all">{this.state.error?.message?.slice(0, 300)}</p>
   549→          <button
   550→            onClick={() => this.setState({ hasError: false, error: null })}
   551→            className="rounded-md border border-border/60 px-4 py-2 text-sm hover:bg-foreground/5"
   552→          >
   553→            Reintentar
   554→          </button>
   555→        </div>
   556→      );
   557→    }
   558→    return this.props.children;
   559→  }
   560→}
