   248→/* =========================================================
   249→ * Helpers
   250→ * =======================================================*/
   251→function timeToMin(t: string): number {
   252→  const [h, m] = t.split(":").map(Number);
   253→  return h * 60 + m;
   254→}
   255→
   256→function tableW(t: RpTable): number {
   257→  if (t.shape === "round") return 84;
   258→  if (t.shape === "square") return 76;
   259→  return t.w ?? 140;
   260→}
   261→function tableH(t: RpTable): number {
   262→  if (t.shape === "round") return 84;
   263→  if (t.shape === "square") return 76;
   264→  return t.h ?? 96;
   265→}
   266→
   267→function shapeLabel(s: RpTable["shape"]): string {
   268→  if (s === "rect") return "rectangular";
   269→  if (s === "round") return "redonda";
   270→  return "cuadrada";
   271→}
   272→
   273→function initials(name: string): string {
   274→  return name
   275→    .split(" ")
   276→    .map((p) => p[0])
   277→    .filter(Boolean)
   278→    .slice(0, 2)
   279→    .join("")
   280→    .toUpperCase();
   281→}

   348→      ts.map((x) => (x.id === t.id ? { ...x, status: next } : x))
   349→    );
   350→    toast({
   351→      title: `Mesa ${t.name}`,

   365→          title: "Mesa no disponible",

   367→          variant: "destructive",

   376→  function assignTableToReservation(tableId: string, reservationId: string) {
   377→    const t = tables.find((x) => x.id === tableId);
   378→    const r = reservations.find((x) => x.id === reservationId);
   379→    if (!t || !r) return;
   380→    setTables((ts) =>
   381→      ts.map((x) =>
   382→        x.id === tableId
   383→          ? { ...x, status: "reserved", reservationId }
   384→          : x
   385→      )
   386→    );
   387→    setReservations((rs) =>
   388→      rs.map((x) =>
   389→        x.id === reservationId
   390→          ? {
   391→              ...x,
   392→              tableId,
   393→              status: x.status === "espera" ? "confirmada" : x.status,
   394→            }
   395→          : x
   396→      )
   397→    );
   398→    toast({
   399→      title: "Mesa asignada",
   400→      description: `${r.customerName} → ${t.name}`,
   401→    });
   402→  }

   417→        rs.map((r) =>
   418→          r.id === prevResId ? { ...r, tableId: undefined } : r

   479→        ts.map((t) =>
   480→          t.id === r.tableId

   499→  /* ----- drag & drop (mouse) ----- */
   500→  function onDragStart(e: React.DragEvent, t: RpTable) {
   501→    if (!editMode) {
   502→      e.preventDefault();
   503→      return;
   504→    }
   505→    e.dataTransfer.setData("text/plain", t.id);
   506→    e.dataTransfer.effectAllowed = "move";
   507→    setDraggedId(t.id);
   508→  }
   509→
   510→  function onDragEnd() {
   511→    setDraggedId(null);
   512→    setDragOver(false);
   513→  }
   514→
   515→  function onCanvasDragOver(e: React.DragEvent) {
   516→    if (!editMode || !draggedId) return;
   517→    e.preventDefault();
   518→    e.dataTransfer.dropEffect = "move";
   519→    if (!dragOver) setDragOver(true);
   520→  }
   521→
   522→  function onCanvasDragLeave(e: React.DragEvent) {
   523→    if (e.currentTarget === e.target) setDragOver(false);
   524→  }
   525→
   526→  function onCanvasDrop(e: React.DragEvent) {
   527→    if (!editMode || !draggedId) return;
   528→    e.preventDefault();
   529→    const rect = canvasRef.current?.getBoundingClientRect();
   530→    if (!rect) return;
   531→    const t = tables.find((x) => x.id === draggedId);
   532→    if (!t) return;
   533→    const w = tableW(t);
   534→    const h = tableH(t);

   557→   * Native HTML5 drag-and-drop does NOT fire on touch devices, so we add
   558→   * touch handlers that move the table in real time when editMode is on. */
   559→  const touchDragRef = React.useRef<
   560→    | { id: string; offsetX: number; offsetY: number }
   561→    | null
   562→  >(null);

   564→  function onTableTouchStart(e: React.TouchEvent, t: RpTable) {
   565→    if (!editMode) return;
   566→    if (e.touches.length !== 1) return;
   567→    const rect = canvasRef.current?.getBoundingClientRect();
   568→    if (!rect) return;
   569→    const touch = e.touches[0];
   570→    touchDragRef.current = {
   571→      id: t.id,
   572→      offsetX: touch.clientX - rect.left - t.x,
   573→      offsetY: touch.clientY - rect.top - t.y,
   574→    };
   575→    setDraggedId(t.id);
   576→    setDragOver(true);
   577→  }
   578→
   579→  function onCanvasTouchMove(e: React.TouchEvent) {
   580→    if (!editMode || !touchDragRef.current) return;
   581→    if (e.touches.length !== 1) return;
   582→    e.preventDefault(); // stop page scroll while dragging
   583→    const rect = canvasRef.current?.getBoundingClientRect();
   584→    if (!rect) return;
   585→    const touch = e.touches[0];
   586→    const drag = touchDragRef.current;
   587→    const t = tables.find((x) => x.id === drag.id);
   588→    if (!t) return;
   589→    const w = tableW(t);
   590→    const h = tableH(t);
   591→    const x = Math.max(
   592→      4,
   593→      Math.min(touch.clientX - rect.left - drag.offsetX, rect.width - w - 4)
   594→    );
   595→    const y = Math.max(
   596→      4,
   597→      Math.min(touch.clientY - rect.top - drag.offsetY, rect.height - h - 4)
   598→    );
   599→    setTables((ts) =>
   600→      ts.map((x2) => (x2.id === drag.id ? { ...x2, x: Math.round(x), y: Math.round(y) } : x2))
   601→    );
   602→  }
   603→
   604→  function onCanvasTouchEnd() {
   605→    if (!touchDragRef.current) return;
   606→    const t = tables.find((x) => x.id === touchDragRef.current!.id);
   607→    touchDragRef.current = null;
   608→    setDraggedId(null);
   609→    setDragOver(false);
   610→    if (t) {
   611→      toast({
   612→        title: `${t.name} reposicionada`,
   613→        description: `x ${t.x} · y ${t.y}`,
   614→      });
   615→    }

  1555→/* =========================================================
  1556→ * Service timeline
  1557→ * =======================================================*/
  1558→function TimelineContent({
  1559→  reservations,
  1560→  tables,
  1561→  selectedId,
  1562→  onSelect,
  1563→}: {
  1564→  reservations: RpReservation[];
  1565→  tables: RpTable[];
  1566→  selectedId: string | null;
  1567→  onSelect: (id: string) => void;
  1568→}) {
  1569→  const START = 13 * 60; // 13:00
  1570→  const END = 23 * 60; // 23:00
  1571→  const PX_PER_MIN = 1.5;
  1572→  const width = (END - START) * PX_PER_MIN;
  1573→
  1574→  const hours: number[] = [];
  1575→  for (let h = 13; h <= 23; h++) hours.push(h);
  1576→
  1577→  // greedy lane packing
  1578→  const lanesEnd: number[] = [];
  1579→  const placed = reservations
  1580→    .filter((r) => r.status !== "cancelada")
  1581→    .map((r) => {
  1582→      const start = timeToMin(r.time);
  1583→      const end = start + r.durationMin;
  1584→      let lane = lanesEnd.findIndex((l) => l <= start);
  1585→      if (lane === -1) {
  1586→        lane = lanesEnd.length;
  1587→        lanesEnd.push(end);
  1588→      } else {
  1589→        lanesEnd[lane] = end;
  1590→      }
  1591→      return { r, start, end, lane };
  1592→    })
  1593→    .sort((a, b) => a.start - b.start);
  1594→
  1595→  const laneCount = Math.max(1, lanesEnd.length);
  1596→  const rowH = 38;
