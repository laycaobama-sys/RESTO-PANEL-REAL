   391→function initials(name: string): string {
   392→  return name
   393→    .split(" ")
   394→    .slice(0, 2)
   395→    .map((p) => p[0])
   396→    .join("")
   397→    .toUpperCase();
   398→}

   426→function TagChip({
   427→  tag,
   428→  onRemove,
   429→  removable = false,
   430→  disabled = false,
   431→}: {
   432→  tag: TagId;
   433→  onRemove?: () => void;
   434→  removable?: boolean;
   435→  disabled?: boolean;
   436→}) {
   437→  const def = ALL_TAGS.find((t) => t.id === tag)!;
   438→  return (
   439→    <span
   440→      className={cn(
   441→        "inline-flex items-center gap-1 rounded-md border px-2 py-0.5 text-[11px] font-mono uppercase tracking-wider",
   442→        TAG_TONE_CLASS[def.tone]
   443→      )}
   444→    >
   445→      {tag === "VIP" && <Crown className="h-3 w-3" aria-hidden />}
   446→      {tag === "cumpleaños" && <Gift className="h-3 w-3" aria-hidden />}
   447→      {tag === "riesgo" && <AlertTriangle className="h-3 w-3" aria-hidden />}
   448→      {def.label}
   449→      {removable && !disabled && (
   450→        <button
   451→          type="button"
   452→          onClick={onRemove}
   453→          className="ml-0.5 rounded-sm hover:bg-foreground/15 focus-visible:outline focus-visible:outline-2 focus-visible:outline-ring"
   454→          aria-label={`Quitar etiqueta ${def.label}`}
   455→        >
   456→          <X className="h-3 w-3" />
   457→        </button>
   458→      )}
   459→    </span>
   460→  );
   461→}
