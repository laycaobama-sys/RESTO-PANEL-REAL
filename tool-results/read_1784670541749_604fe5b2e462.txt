   256→   256→function tableW(t: RpTable): number {
   257→   257→  if (t.shape === "round") return 84;
   258→   258→  if (t.shape === "square") return 76;
   259→   259→  return t.w ?? 140;
   260→   260→}
   261→   261→function tableH(t: RpTable): number {
   262→   262→  if (t.shape === "round") return 84;
   263→   263→  if (t.shape === "square") return 76;
   264→   264→  return t.h ?? 96;
   265→   265→}

   499→   499→  /* ----- drag & drop ----- */
   500→   500→  function onDragStart(e: React.DragEvent, t: RpTable) {
   501→   501→    if (!editMode) {
   502→   502→      e.preventDefault();
   503→   503→      return;
   504→   504→    }
   505→   505→    e.dataTransfer.setData("text/plain", t.id);
   506→   506→    e.dataTransfer.effectAllowed = "move";
   507→   507→    setDraggedId(t.id);
   508→   508→  }

   510→   510→  function onDragEnd() {
   511→   511→    setDraggedId(null);
   512→   512→    setDragOver(false);

   515→   515→  function onCanvasDragOver(e: React.DragEvent) {
   516→   516→    if (!editMode || !draggedId) return;
   517→   517→    e.preventDefault();
   518→   518→    e.dataTransfer.dropEffect = "move";
   519→   519→    if (!dragOver) setDragOver(true);
   520→   520→  }
   521→   521→
   522→   522→  function onCanvasDragLeave(e: React.DragEvent) {
   523→   523→    if (e.currentTarget === e.target) setDragOver(false);
   524→   524→  }

   526→   526→  function onCanvasDrop(e: React.DragEvent) {
   527→   527→    if (!editMode || !draggedId) return;
   528→   528→    e.preventDefault();
   529→   529→    const rect = canvasRef.current?.getBoundingClientRect();
   530→   530→    if (!rect) return;
   531→   531→    const t = tables.find((x) => x.id === draggedId);
   532→   532→    if (!t) return;
   533→   533→    const w = tableW(t);
   534→   534→    const h = tableH(t);
   535→   535→    const x = Math.max(
   536→   536→      4,
   537→   537→      Math.min(e.clientX - rect.left - w / 2, rect.width - w - 4)
   538→   538→    );
   539→   539→    const y = Math.max(
   540→   540→      4,
   541→   541→      Math.min(e.clientY - rect.top - h / 2, rect.height - h - 4)
   542→   542→    );
   543→   543→    const nx = Math.round(x);
   544→   544→    const ny = Math.round(y);
   545→   545→    setTables((ts) =>
   546→   546→      ts.map((x2) => (x2.id === draggedId ? { ...x2, x: nx, y: ny } : x2))
   547→   547→    );
   548→   548→    setDragOver(false);
   549→   549→    setDraggedId(null);
   550→   550→    toast({
   551→   551→      title: `${t.name} reposicionada`,
   552→   552→      description: `x ${nx} · y ${ny}`,
   553→   553→    });
   554→   554→  }
