   591→export function ExecAi() {
   592→  const reduceMotion = useReducedMotion();
   593→  const [messages, setMessages] = React.useState<AiMessage[]>(INITIAL_MESSAGES);
   594→  const [input, setInput] = React.useState("");
   595→  const [isTyping, setIsTyping] = React.useState(false);
   596→  const scrollRef = React.useRef<HTMLDivElement>(null);
   597→  const inputRef = React.useRef<HTMLInputElement>(null);
   598→
   599→  // Auto-scroll to bottom on new message
   600→  React.useEffect(() => {
   601→    const el = scrollRef.current;
   602→    if (el) {
   603→      el.scrollTo({ top: el.scrollHeight, behavior: reduceMotion ? "auto" : "smooth" });
   604→    }
   605→  }, [messages, isTyping, reduceMotion]);
   606→
   607→  function send(text: string) {
   608→    const trimmed = text.trim();
   609→    if (!trimmed || isTyping) return;
   610→    const now = new Date();
   611→    const ts = `${String(now.getHours()).padStart(2, "0")}:${String(now.getMinutes()).padStart(2, "0")}`;
   612→    const userMsg: AiMessage = {
   613→      id: `u-${Date.now()}`,
   614→      role: "user",
   615→      content: trimmed,
   616→      ts,
   617→    };
   618→    setMessages((m) => [...m, userMsg]);
   619→    setInput("");
   620→    setIsTyping(true);
   621→
   622→    // Simulate AI thinking + response
   623→    window.setTimeout(() => {
   624→      const demo = pickDemoResponse();

   626→        id: `a-${Date.now()}`,
   627→        role: "ai",
   628→        ts,
   629→        ...demo,

   633→    }, 1400);
   634→  }

   654→                  IA Ejecutiva
   655→                </h2>
   656→                <span className="text-[10px] font-mono uppercase tracking-wider px-1.5 py-0.5 rounded border border-amber-400/40 bg-amber-400/10 text-amber-300">
   657→                  demo
   658→                </span>
   659→              </div>

   662→              </div>
   663→            </div>

   665→          <div className="hidden sm:flex items-center gap-1.5 text-[11px] font-mono text-[var(--teal)] shrink-0">
