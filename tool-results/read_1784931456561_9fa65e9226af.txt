   309→interface CmdAction {
   310→  id: string;
   311→  label: string;
   312→  hint: string;
   313→  icon: React.ElementType;
   314→  group: "Navegación" | "Acciones";
   315→  run: () => void;
   316→}
   317→
   318→function CommandPalette({ open, onOpenChange }: { open: boolean; onOpenChange: (o: boolean) => void }) {
   319→  const [query, setQuery] = React.useState("");
   320→  const [active, setActive] = React.useState(0);
   321→  const inputRef = React.useRef<HTMLInputElement>(null);
   322→  const listRef = React.useRef<HTMLDivElement>(null);
   323→
   324→  const actions = React.useMemo<CmdAction[]>(() => {
   325→    const nav: CmdAction[] = NAV.map((n) => ({
   326→      id: `nav-${n.id}`,
   327→      label: n.label,
   328→      hint: "Ir a sección",
   329→      icon: n.icon,
   330→      group: "Navegación",
   331→      run: () => useNav.getState().go(n.id),
   332→    }));
   333→    const quick: CmdAction[] = [
   334→      { id: "qa-executive", label: "Abrir Centro Ejecutivo", hint: "IA + Alertas", icon: Crown, group: "Acciones", run: () => useNav.getState().go("executive") },
   335→      { id: "qa-new-reserva", label: "Nueva reserva", hint: "Crear reserva", icon: CalendarPlus, group: "Acciones", run: () => useNav.getState().go("reservas") },
   336→      { id: "qa-search-client", label: "Buscar cliente", hint: "Abrir CRM", icon: UserSearch, group: "Acciones", run: () => useNav.getState().go("crm") },
   337→      { id: "qa-floor-plan", label: "Ver plano de mesas", hint: "Abrir reservas", icon: MapIcon, group: "Acciones", run: () => useNav.getState().go("reservas") },
   338→      { id: "qa-marketing", label: "Lanzar campaña", hint: "Marketing", icon: Megaphone, group: "Acciones", run: () => useNav.getState().go("marketing") },
   339→      { id: "qa-growth-analytics", label: "Ver Growth Analytics", hint: "ROI e insights IA", icon: TrendingUp, group: "Acciones", run: () => useNav.getState().go("growth-analytics") },
   340→      { id: "qa-growth-reputation", label: "Gestionar reputación", hint: "Reseñas + NPS", icon: Award, group: "Acciones", run: () => useNav.getState().go("growth-reputation") },
   341→      { id: "qa-reviews", label: "Responder reviews", hint: "Google Reviews", icon: Star, group: "Acciones", run: () => useNav.getState().go("reviews") },
   342→      { id: "qa-analytics", label: "Ver analítica", hint: "Analytics", icon: BarChart3, group: "Acciones", run: () => useNav.getState().go("analytics") },
   343→      { id: "qa-settings", label: "Configuración", hint: "Ajustes del local", icon: Settings, group: "Acciones", run: () => useNav.getState().go("configuracion") },
   344→    ];
   345→    return [...quick, ...nav];
   346→  }, []);
   347→
   348→  const filtered = React.useMemo(() => {
   349→    const q = query.trim().toLowerCase();
   350→    if (!q) return actions;
   351→    return actions.filter((a) =>
   352→      a.label.toLowerCase().includes(q) || a.hint.toLowerCase().includes(q)
   353→    );
   354→  }, [actions, query]);
   355→
   356→  // Reset state on open
   357→  React.useEffect(() => {
   358→    if (open) {
   359→      setQuery("");
   360→      setActive(0);
   361→      // focus input after dialog mounts
   362→      requestAnimationFrame(() => inputRef.current?.focus());
   363→    }
   364→  }, [open]);
   365→
   366→  // Keep active index in range
   367→  React.useEffect(() => {
   368→    if (active >= filtered.length) setActive(0);
   369→  }, [filtered, active]);
   370→
   371→  // Scroll active item into view
   372→  React.useEffect(() => {
   373→    const el = listRef.current?.querySelector<HTMLElement>(`[data-idx="${active}"]`);
   374→    el?.scrollIntoView({ block: "nearest" });
   375→  }, [active]);
   376→
   377→  function execute(a: CmdAction) {
   378→    a.run();
   379→    onOpenChange(false);
   380→  }
   381→
   382→  function onKeyDown(e: React.KeyboardEvent) {
   383→    if (e.key === "ArrowDown") {
   384→      e.preventDefault();
   385→      setActive((i) => (i + 1) % Math.max(filtered.length, 1));
   386→    } else if (e.key === "ArrowUp") {
   387→      e.preventDefault();
   388→      setActive((i) => (i - 1 + filtered.length) % Math.max(filtered.length, 1));
   389→    } else if (e.key === "Enter") {
   390→      e.preventDefault();
   391→      const a = filtered[active];
   392→      if (a) execute(a);
   393→    } else if (e.key === "Escape") {
   394→      onOpenChange(false);
   395→    }
   396→  }
   397→
   398→  // Group filtered actions for rendering
   399→  const groups = React.useMemo(() => {
   400→    const map = new Map<string, CmdAction[]>();
   401→    for (const a of filtered) {
   402→      const arr = map.get(a.group) ?? [];
   403→      arr.push(a);
   404→      map.set(a.group, arr);
   405→    }
   406→    return Array.from(map.entries());
   407→  }, [filtered]);
   408→
   409→  // Flat index lookup for keyboard nav
   410→  let flatIdx = -1;
   411→
   412→  return (
   413→    <Dialog open={open} onOpenChange={onOpenChange}>

   544→class ErrorBoundary extends React.Component<
   545→  { children: React.ReactNode },
   546→  { hasError: boolean; error: Error | null }
   547→> {
   548→  constructor(props: { children: React.ReactNode }) {
   549→    super(props);
   550→    this.state = { hasError: false, error: null };
   551→  }
   552→  static getDerivedStateFromError(error: Error) {
   553→    return { hasError: true, error };
   554→  }
   555→  componentDidCatch(error: Error) {
   556→    console.error("SectionRenderer error:", error.message);
   557→  }
   558→  render() {
   559→    if (this.state.hasError) {
   560→      return (
   561→        <div className="rp-glass rounded-xl p-8 text-center max-w-2xl mx-auto">
   562→          <p className="text-sm text-destructive mb-3 font-medium">Error al cargar esta sección</p>
   563→          <p className="text-xs text-muted-foreground mb-4 font-mono break-all">{this.state.error?.message?.slice(0, 300)}</p>
   564→          <button
   565→            onClick={() => this.setState({ hasError: false, error: null })}
   566→            className="rounded-md border border-border/60 px-4 py-2 text-sm hover:bg-foreground/5"
   567→          >
   568→            Reintentar
   569→          </button>
   570→        </div>
   571→      );
   572→    }
   573→    return this.props.children;
   574→  }
   575→}
