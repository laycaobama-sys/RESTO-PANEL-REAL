   248→/* =========================================================
   249→ * Helpers
   250→ * =======================================================*/
   251→function timeToMin(t: string): number {
   252→  const [h, m] = t.split(":").map(Number);
   253→  return h * 60 + m;
   254→}
   255→
   256→function tableW(t: RpTable): number {
   257→  if (t.shape === "round") return 84;
   258→  if (t.shape === "square") return 76;
   259→  return t.w ?? 140;
   260→}
   261→function tableH(t: RpTable): number {
   262→  if (t.shape === "round") return 84;
   263→  if (t.shape === "square") return 76;
   264→  return t.h ?? 96;
   265→}
   266→
   267→function shapeLabel(s: RpTable["shape"]): string {
   268→  if (s === "rect") return "rectangular";
   269→  if (s === "round") return "redonda";
   270→  return "cuadrada";
   271→}
   272→
   273→function initials(name: string): string {
   274→  return name
   275→    .split(" ")
   276→    .map((p) => p[0])
   277→    .filter(Boolean)
   278→    .slice(0, 2)
   279→    .join("")
   280→    .toUpperCase();
   281→}

   348→      ts.map((x) => (x.id === t.id ? { ...x, status: next } : x))
   349→    );
   350→    toast({
   351→      title: `Mesa ${t.name}`,

   365→          title: "Mesa no disponible",

   376→  function assignTableToReservation(tableId: string, reservationId: string) {
   377→    const t = tables.find((x) => x.id === tableId);

   379→    if (!t || !r) return;
   380→    setTables((ts) =>
   381→      ts.map((x) =>
   382→        x.id === tableId
   383→          ? { ...x, status: "reserved", reservationId }
   384→          : x
   385→      )
   386→    );
   387→    setReservations((rs) =>
   388→      rs.map((x) =>
   389→        x.id === reservationId
   390→          ? {
   391→              ...x,
   392→              tableId,
   393→              status: x.status === "espera" ? "confirmada" : x.status,
   394→            }
   395→          : x
   396→      )
   397→    );
   398→    toast({
   399→      title: "Mesa asignada",
   400→      description: `${r.customerName} → ${t.name}`,
   401→    });
   402→  }

   417→        rs.map((r) =>
   418→          r.id === prevResId ? { ...r, tableId: undefined } : r

   479→        ts.map((t) =>
   480→          t.id === r.tableId

   499→  /* ----- drag & drop ----- */
   500→  function onDragStart(e: React.DragEvent, t: RpTable) {
   501→    if (!editMode) {
   502→      e.preventDefault();
   503→      return;
   504→    }
   505→    e.dataTransfer.setData("text/plain", t.id);
   506→    e.dataTransfer.effectAllowed = "move";
   507→    setDraggedId(t.id);
   508→  }
   509→
   510→  function onDragEnd() {
   511→    setDraggedId(null);
   512→    setDragOver(false);
   513→  }
   514→
   515→  function onCanvasDragOver(e: React.DragEvent) {
   516→    if (!editMode || !draggedId) return;
   517→    e.preventDefault();
   518→    e.dataTransfer.dropEffect = "move";
   519→    if (!dragOver) setDragOver(true);
   520→  }
   521→
   522→  function onCanvasDragLeave(e: React.DragEvent) {
   523→    if (e.currentTarget === e.target) setDragOver(false);
   524→  }
   525→
   526→  function onCanvasDrop(e: React.DragEvent) {
   527→    if (!editMode || !draggedId) return;
   528→    e.preventDefault();
   529→    const rect = canvasRef.current?.getBoundingClientRect();
   530→    if (!rect) return;
   531→    const t = tables.find((x) => x.id === draggedId);
   532→    if (!t) return;
   533→    const w = tableW(t);
   534→    const h = tableH(t);

  1487→/* =========================================================
  1488→ * Service timeline

  1490→function TimelineContent({
  1491→  reservations,
  1492→  tables,
  1493→  selectedId,
  1494→  onSelect,
  1495→}: {
  1496→  reservations: RpReservation[];
  1497→  tables: RpTable[];
  1498→  selectedId: string | null;
  1499→  onSelect: (id: string) => void;
  1500→}) {
  1501→  const START = 13 * 60; // 13:00
  1502→  const END = 23 * 60; // 23:00
  1503→  const PX_PER_MIN = 1.5;
  1504→  const width = (END - START) * PX_PER_MIN;
  1505→
  1506→  const hours: number[] = [];
  1507→  for (let h = 13; h <= 23; h++) hours.push(h);
  1508→
  1509→  // greedy lane packing
  1510→  const lanesEnd: number[] = [];
  1511→  const placed = reservations
  1512→    .filter((r) => r.status !== "cancelada")
  1513→    .map((r) => {
  1514→      const start = timeToMin(r.time);
  1515→      const end = start + r.durationMin;
  1516→      let lane = lanesEnd.findIndex((l) => l <= start);
  1517→      if (lane === -1) {
  1518→        lane = lanesEnd.length;
  1519→        lanesEnd.push(end);
  1520→      } else {
  1521→        lanesEnd[lane] = end;
  1522→      }
  1523→      return { r, start, end, lane };
  1524→    })
  1525→    .sort((a, b) => a.start - b.start);
  1526→
  1527→  const laneCount = Math.max(1, lanesEnd.length);
  1528→  const rowH = 38;
  1529→  const totalH = 26 + laneCount * rowH + 8;
  1530→
  1531→  // mock "now" indicator at 14:45
  1532→  const nowMin = 14 * 60 + 45;
  1533→  const nowX = (nowMin - START) * PX_PER_MIN;
  1534→  const showNow = nowMin >= START && nowMin <= END;
  1535→
  1536→  return (
  1537→    <div className="overflow-x-auto rp-scroll-thin -mx-1">
  1538→      <div
  1539→        className="relative"
  1540→        style={{ width, minWidth: "100%", height: totalH }}
  1541→      >
  1542→        {/* hour grid */}
  1543→        {hours.map((h) => {
  1544→          const x = (h * 60 - START) * PX_PER_MIN;
  1545→          return (
  1546→            <div
  1547→              key={h}
  1548→              className="absolute top-0 bottom-0 border-l border-border/30"
  1549→              style={{ left: x }}
  1550→            >
  1551→              <span className="absolute top-0 left-1 text-[10px] font-mono text-muted-foreground">
  1552→                {String(h).padStart(2, "0")}:00
  1553→              </span>
  1554→            </div>
  1555→          );
  1556→        })}
  1557→
  1558→        {/* now indicator */}
  1559→        {showNow && (
  1560→          <div
  1561→            className="absolute top-0 bottom-0 w-px bg-[var(--teal)] z-10"
  1562→            style={{ left: nowX }}
  1563→            aria-hidden
  1564→          >
  1565→            <div className="absolute top-0 -translate-x-1/2 h-2 w-2 rounded-full bg-[var(--teal)]" />
  1566→            <div className="absolute top-0 -translate-x-1/2 -translate-y-full text-[9px] font-mono text-[var(--teal)] whitespace-nowrap bg-background/80 px-1 rounded">
  1567→              AHORA
  1568→            </div>
  1569→          </div>
  1570→        )}
  1571→
  1572→        {/* reservation blocks */}
  1573→        {placed.map(({ r, start, end, lane }) => {
  1574→          const x = (start - START) * PX_PER_MIN;
  1575→          const w = (end - start) * PX_PER_MIN;
  1576→          const y = 22 + lane * rowH;
  1577→          const table = tables.find((t) => t.id === r.tableId);
  1578→          const isSelected = selectedId === r.id;
  1579→          const meta = RES_STATUS_META[r.status];
  1580→          return (
  1581→            <button
  1582→              key={r.id}
  1583→              onClick={() => onSelect(r.id)}
  1584→              aria-pressed={isSelected}
  1585→              aria-label={`${r.customerName} · ${r.time} · ${r.partySize} comensales${
  1586→                table ? ` · mesa ${table.name}` : ""
  1587→              }`}
  1588→              className={cn(
  1589→                "absolute h-8 rounded-md border px-2 text-left text-[11px] flex items-center gap-1.5 overflow-hidden transition-all",
  1590→                meta.cls,
  1591→                isSelected
  1592→                  ? "ring-2 ring-[var(--gold)] z-10 brightness-110"
  1593→                  : "hover:brightness-125"
  1594→              )}
  1595→              style={{ left: x, width: Math.max(w - 2, 60), top: y }}
  1596→              title={`${r.customerName} · ${r.time} · ${r.partySize} pax${
  1597→                table ? ` · ${table.name}` : ""
