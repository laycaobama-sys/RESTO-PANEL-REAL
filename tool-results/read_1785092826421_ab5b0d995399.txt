   443→function Sparkline({
   444→  data,
   445→  color = "var(--gold)",
   446→  width = 96,
   447→  height = 28,
   448→  fill = true,
   449→}: {
   450→  data: number[];
   451→  color?: string;
   452→  width?: number;
   453→  height?: number;
   454→  fill?: boolean;
   455→}) {
   456→  const id = React.useId();
   457→  if (!data.length) return null;
   458→  const max = Math.max(...data);
   459→  const min = Math.min(...data);
   460→  const range = max - min || 1;
   461→  const step = width / (data.length - 1 || 1);
   462→  const pts = data.map((d, i) => {
   463→    const x = i * step;
   464→    const y = height - 2 - ((d - min) / range) * (height - 4);
   465→    return [x, y] as const;
   466→  });
   467→  const linePath = pts.map((p, i) => (i === 0 ? `M${p[0]},${p[1]}` : `L${p[0]},${p[1]}`)).join(" ");
   468→  const areaPath = `${linePath} L${width},${height} L0,${height} Z`;
   469→  return (
   470→    <svg viewBox={`0 0 ${width} ${height}`} className="w-full" preserveAspectRatio="none" aria-hidden>
   471→      {fill && (
   472→        <>
   473→          <defs>
   474→            <linearGradient id={`spark-${id}`} x1="0" y1="0" x2="0" y2="1">
   475→              <stop offset="0%" stopColor={color} stopOpacity="0.35" />
   476→              <stop offset="100%" stopColor={color} stopOpacity="0" />
   477→            </linearGradient>
   478→          </defs>
   479→          <path d={areaPath} fill={`url(#spark-${id})`} />
   480→        </>
   481→      )}
   482→      <path d={linePath} fill="none" stroke={color} strokeWidth="1.5" strokeLinecap="round" strokeLinejoin="round" />
   483→      <circle cx={pts[pts.length - 1][0]} cy={pts[pts.length - 1][1]} r="1.6" fill={color} />
   484→    </svg>
   485→  );
   486→}

   488→function Gauge({ value, max = 100 }: { value: number; max?: number }) {
   489→  const pct = Math.max(0, Math.min(1, value / max));
   490→  const r = 38;

   494→  const dash = circ * pct;

   496→    <svg viewBox="0 0 96 60" className="w-full max-w-[140px]" role="img" aria-label={`Ocupación ${value}%`}>
   497→      <path

   499→        fill="none"
   500→        stroke="currentColor"
   501→        className="text-foreground/10"
   502→        strokeWidth="8"
   503→        strokeLinecap="round"

   507→        fill="none"

   530→function Donut({
   531→  segments,
   532→  size = 80,
   533→}: {
   534→  segments: { label: string; value: number; color: string }[];
   535→  size?: number;
   536→}) {
   537→  const r = 14;
   538→  const cx = 16;
   539→  const cy = 16;
   540→  const circ = 2 * Math.PI * r;
   541→  const lens = segments.map((s) => (s.value / 100) * circ);
   542→  const cumOffsets = lens.map((_, i) => -lens.slice(0, i).reduce((a, b) => a + b, 0));
   543→  return (
   544→    <svg viewBox="0 0 32 32" width={size} height={size} role="img" aria-label="Distribución por canal">
   545→      <circle cx={cx} cy={cy} r={r} fill="none" stroke="currentColor" className="text-foreground/8" strokeWidth="6" />
   546→      {segments.map((s, i) => {
   547→        const len = lens[i];
   548→        const offset = cumOffsets[i];
   549→        return (
   550→          <circle
   551→            key={i}
   552→            cx={cx}
   553→            cy={cy}
   554→            r={r}
   555→            fill="none"
   556→            stroke={s.color}
   557→            strokeWidth="6"
   558→            strokeDasharray={`${len} ${circ - len}`}
   559→            strokeDashoffset={offset}
   560→            transform={`rotate(-90 ${cx} ${cy})`}
   561→            strokeLinecap="butt"
   562→          />
   563→        );
   564→      })}
   565→    </svg>
   566→  );
   567→}

  1337→function ForecastChart() {
  1338→  const width = 720;
  1339→  const height = 280;
  1340→  const padL = 44;
  1341→  const padR = 16;
  1342→  const padT = 24;
  1343→  const padB = 36;
  1344→  const innerW = width - padL - padR;
  1345→  const innerH = height - padT - padB;
  1346→  const allVals = [142, 138, 158, 172, 188, 196, 184, 168, 174, 178, 182, 186, 198, 206, 204, 210, 168, 174, 192];
  1347→  const maxV = Math.max(...allVals) * 1.05;
  1348→  const minV = Math.min(...allVals) * 0.9;
  1349→
  1350→  const days = FORECAST_DAYS;
  1351→  const n = days.length;
  1352→  const step = innerW / (n - 1);
  1353→
  1354→  const yOf = (v: number) => padT + innerH - ((v - minV) / (maxV - minV)) * innerH;
  1355→  const xOf = (i: number) => padL + i * step;
  1356→
  1357→  const actualPts = FORECAST_ACTUAL.map((v, i) => [xOf(i), yOf(v)] as const);
  1358→  const actualPath = actualPts.map((p, i) => (i === 0 ? `M${p[0]},${p[1]}` : `L${p[0]},${p[1]}`)).join(" ");
  1359→
  1360→  // Forecast line: starts from last actual point
  1361→  const fIdx = [n - 1, n, n + 1]; // tomorrow onwards — we'll just show 1 forecast point with band
  1362→  // Simpler: forecast line connects last actual to a future value to the right edge
  1363→  const forecastX = width - padR;
  1364→  const forecastY = yOf(186);
  1365→  const forecastPath = `M${actualPts[n - 1][0]},${actualPts[n - 1][1]} L${forecastX},${forecastY}`;
  1366→
  1367→  // Confidence band: triangle from last actual point expanding
  1368→  const bandPath = `M${actualPts[n - 1][0]},${actualPts[n - 1][1]} L${forecastX},${yOf(168)} L${forecastX},${yOf(204)} Z`;
  1369→
  1370→  const yTicks = [minV, (minV + maxV) / 2, maxV];
  1371→
  1372→  return (
  1373→    <div className="overflow-x-auto rp-scroll-thin">
  1374→      <svg viewBox={`0 0 ${width} ${height}`} className="w-full min-w-[640px]" role="img" aria-label="Forecast 7 días con banda de confianza">

  1376→        {yTicks.map((t, i) => {
  1377→          const y = yOf(t);
  1378→          return (
  1379→            <g key={i}>
  1380→              <line x1={padL} x2={width - padR} y1={y} y2={y} stroke="currentColor" className="text-foreground/8" strokeDasharray="2 4" />
  1381→              <text x={padL - 8} y={y + 3} textAnchor="end" className="fill-muted-foreground font-mono" fontSize="9">
  1382→                {Math.round(t)}
  1383→              </text>
  1384→            </g>
  1385→          );
  1386→        })}
  1387→        {/* x labels */}
  1388→        {days.map((d, i) => (
  1389→          <text key={d} x={xOf(i)} y={height - 12} textAnchor="middle" className="fill-muted-foreground font-mono" fontSize="9">
  1390→            {d}
  1391→          </text>
  1392→        ))}

  1394→          Mañana
  1395→        </text>
  1396→
  1397→        {/* Confidence band */}
  1398→        <path d={bandPath} fill="var(--teal)" fillOpacity="0.12" stroke="none" />
  1399→
  1400→        {/* Actual line (gold) */}
  1401→        <path d={actualPath} fill="none" stroke="var(--gold)" strokeWidth="2.2" strokeLinecap="round" strokeLinejoin="round" />
  1402→        {actualPts.map((p, i) => (

  1404→        ))}

  1406→        {/* Forecast line (dashed teal) */}

  1806→function heatValue(metricId: string, day: number, hour: number): number {
  1807→  let v = 0;
  1808→  // Lunch peak 13-15
  1809→  v += Math.exp(-Math.pow(hour - 14, 2) / 6) * 0.8;
  1810→  // Dinner peak 20-22
  1811→  v += Math.exp(-Math.pow(hour - 21, 2) / 5) * 0.95;
  1812→  // Late night / early morning floor
  1813→  if (hour < 11) v *= 0.15;
  1814→  if (hour >= 0 && hour <= 5) v = 0.04 + Math.random() * 0.02;
  1815→  // Weekend bonus
  1816→  const weekend = day >= 5 ? 0.18 : 0;
  1817→  // Tuesday dip
  1818→  const tueDip = day === 1 ? -0.28 : 0;
  1819→  // Thursday terraza peak
  1820→  const thuPeak = day === 3 ? 0.12 : 0;
  1821→  // Sunday ticket dip
  1822→  const sunDip = day === 6 ? -0.05 : 0;
  1823→  v = v + weekend + tueDip + thuPeak + sunDip;
  1824→
  1825→  if (metricId === "noshows" || metricId === "cancelaciones") {
  1826→    v = v * 0.5 + (hour >= 19 ? 0.55 : 0.12) + (day >= 5 ? 0.12 : 0) + Math.sin(day * 0.7) * 0.06;
  1827→  }
  1828→  if (metricId === "ticket") {
  1829→    v = v * 0.55 + (hour >= 19 ? 0.32 : 0.15) + (day === 6 ? -0.1 : 0.05);
  1830→  }
  1831→  if (metricId === "vip") {
  1832→    v = v * 0.35 + (hour >= 20 ? 0.45 : 0.06) + (day >= 5 ? 0.15 : 0);
  1833→  }
  1834→  if (metricId === "rotacion") {
  1835→    v = v * 1.1;
  1836→  }
  1837→  if (metricId === "ingresos") {
  1838→    v = v * 0.9 + (hour >= 19 ? 0.2 : 0.05);
  1839→  }
  1840→
  1841→  return Math.max(0, Math.min(1, v));
  1842→}
