   546→   546→/* ---------- Sparkline ---------- */
   547→   547→
   548→   548→function Sparkline({ data, color }: { data: number[]; color: "gold" | "teal" }) {
   549→   549→  const rawId = React.useId();
   550→   550→  const id = rawId.replace(/[:]/g, "");
   551→   551→  const w = 76;
   552→   552→  const h = 28;
   553→   553→  const pad = 3;
   554→   554→  const max = Math.max(...data);
   555→   555→  const min = Math.min(...data);
   556→   556→  const range = max - min || 1;
   557→   557→  const pts = data.map((v, i) => {
   558→   558→    const x = (i / (data.length - 1)) * (w - pad * 2) + pad;
   559→   559→    const y = h - pad - ((v - min) / range) * (h - pad * 2);
   560→   560→    return [x, y] as [number, number];
   561→   561→  });
   562→   562→  const linePts = pts.map(([x, y]) => `${x.toFixed(1)},${y.toFixed(1)}`).join(" ");
   563→   563→  const areaPath = `M${pts[0][0].toFixed(1)},${(h - pad).toFixed(1)} L${pts
   564→   564→    .map(([x, y]) => `${x.toFixed(1)},${y.toFixed(1)}`)
   565→   565→    .join(" L")} L${pts[pts.length - 1][0].toFixed(1)},${(h - pad).toFixed(1)} Z`;
   566→   566→  const stroke = color === "gold" ? "var(--gold)" : "var(--teal)";
   567→   567→  const lastPt = pts[pts.length - 1];
   568→   568→
   569→   569→  // Entrance animation: viewport-triggered, once-only, reduced-motion respected.
   570→   570→  const { ref: viewRef, inView } = useInView<SVGSVGElement>({ threshold: 0.2 });
   571→   571→  const progress = useEntranceProgress(inView, 700);
   572→   572→  const { ref: lineRef, length } = usePathLength<SVGPolylineElement>();
   573→   573→  const { dasharray, dashoffset } = drawDash(length, progress);
   574→   574→
   575→   575→  return (
   576→   576→    <svg
   577→   577→      ref={viewRef}
   578→   578→      width={w}
   579→   579→      height={h}
   580→   580→      viewBox={`0 0 ${w} ${h}`}
   581→   581→      aria-hidden="true"
   582→   582→      focusable="false"
   583→   583→      className="shrink-0 overflow-visible max-w-full h-auto"
   584→   584→      preserveAspectRatio="xMidYMid meet"
   585→   585→    >
   586→   586→      <defs>
   587→   587→        <linearGradient id={`spark-${id}`} x1="0" x2="0" y1="0" y2="1">
   588→   588→          <stop offset="0%" stopColor={stroke} stopOpacity="0.32" />
   589→   589→          <stop offset="100%" stopColor={stroke} stopOpacity="0" />
   590→   590→        </linearGradient>
   591→   591→      </defs>
   592→   592→      <path d={areaPath} fill={`url(#spark-${id})`} style={{ opacity: progress }} />
   593→   593→      <polyline
   594→   594→        ref={lineRef}
   595→   595→        points={linePts}
   596→   596→        fill="none"
   597→   597→        stroke={stroke}
   598→   598→        strokeWidth="1.5"
   599→   599→        strokeLinejoin="round"
   600→   600→        strokeLinecap="round"
   601→   601→        strokeDasharray={dasharray}
   602→   602→        strokeDashoffset={dashoffset}
   603→   603→      />
   604→   604→      <circle
   605→   605→        cx={lastPt[0]}
   606→   606→        cy={lastPt[1]}
   607→   607→        r="2"
   608→   608→        fill={stroke}
   609→   609→        style={{ opacity: progress, transition: "opacity 200ms ease-out" }}
   610→   610→      />
   611→   611→    </svg>
   612→   612→  );
   613→   613→}
