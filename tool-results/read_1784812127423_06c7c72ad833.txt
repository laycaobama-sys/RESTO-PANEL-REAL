   448→function statusMeta(s: ReviewStatus): { label: string; cls: string } {
   449→  switch (s) {
   450→    case "new":
   451→      return { label: "Nueva", cls: "border-amber-400/40 bg-amber-400/10 text-amber-300" };
   452→    case "replied":
   453→      return { label: "Respondida", cls: "border-emerald-400/40 bg-emerald-400/10 text-emerald-300" };
   454→    case "pending_approval":
   455→      return { label: "Pendiente aprobación", cls: "border-[var(--gold)]/40 bg-[var(--gold)]/10 text-[var(--gold-soft)]" };
   456→    case "escalated":
   457→      return { label: "Escalada", cls: "border-destructive/40 bg-destructive/10 text-destructive" };
   458→  }
   459→}
