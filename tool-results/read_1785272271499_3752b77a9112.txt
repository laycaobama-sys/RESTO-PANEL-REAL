   214→interface SignalTypeMeta {
   215→  id: SignalType;
   216→  label: string;
   217→  icon: React.ElementType;
   218→  tone: "warn" | "danger" | "teal" | "gold" | "muted";
   219→}

   221→const SIGNAL_TYPES: SignalTypeMeta[] = [
   222→  { id: "limit-80", label: "Límite al 80%", icon: Gauge, tone: "warn" },
   223→  { id: "limit-90", label: "Límite al 90%", icon: Gauge, tone: "danger" },
   224→  { id: "limit-100", label: "Límite al 100%", icon: Ban, tone: "danger" },
   225→  { id: "feature-access", label: "Acceso a feature", icon: Lock, tone: "gold" },
   226→  { id: "growth-pattern", label: "Patrón de crecimiento", icon: TrendingUp, tone: "teal" },
   227→  { id: "volume-increase", label: "Aumento de volumen", icon: Activity, tone: "teal" },
   228→  { id: "plan-superior", label: "Uso de plan superior", icon: Crown, tone: "gold" },

   231→function signalTypeMeta(type: SignalType): SignalTypeMeta {
   232→  return SIGNAL_TYPES.find((s) => s.id === type) ?? SIGNAL_TYPES[0];

   234→
   235→function toneCls(tone: SignalTypeMeta["tone"]): {
   236→  text: string;
   237→  bg: string;
   238→  border: string;
   239→  dot: string;
   240→} {
   241→  switch (tone) {
   242→    case "danger":
   243→      return {
   244→        text: "text-rose-300",
   245→        bg: "bg-rose-500/10",
   246→        border: "border-rose-500/40",
   247→        dot: "bg-rose-500",
   248→      };
   249→    case "warn":
   250→      return {
   251→        text: "text-amber-300",
   252→        bg: "bg-amber-400/10",
   253→        border: "border-amber-400/40",
   254→        dot: "bg-amber-400",
   255→      };
   256→    case "teal":
   257→      return {
   258→        text: "text-[var(--teal)]",
   259→        bg: "bg-[var(--teal)]/10",
   260→        border: "border-[var(--teal)]/40",
   261→        dot: "bg-[var(--teal)]",
   262→      };
   263→    case "gold":
   264→      return {
   265→        text: "text-[var(--gold-soft)]",
   266→        bg: "bg-[var(--gold)]/10",
   267→        border: "border-[var(--gold)]/40",
   268→        dot: "bg-[var(--gold)]",
   269→      };
   270→    default:
   271→      return {
   272→        text: "text-muted-foreground",
   273→        bg: "bg-foreground/[0.04]",
   274→        border: "border-border/40",
   275→        dot: "bg-zinc-500",
   276→      };
   277→  }
   278→}

   337→function formatDate(iso: string): string {
   338→  try {
   339→    const d = new Date(iso);
   340→    return d.toLocaleDateString("es-ES", { day: "2-digit", month: "short", year: "numeric" });
   341→  } catch {
   342→    return iso;
   343→  }
   344→}
   345→
   346→function formatDateTime(iso: string): string {
   347→  try {
   348→    const d = new Date(iso);
   349→    return d.toLocaleString("es-ES", {
   350→      day: "2-digit",
   351→      month: "short",
   352→      hour: "2-digit",
   353→      minute: "2-digit",
   354→    });
   355→  } catch {
   356→    return iso;
   357→  }
   358→}
   359→
   360→function pct(current: number, limit: number): number {
   361→  if (limit <= 0) return 0;
   362→  return Math.min(100, Math.round((current / limit) * 100));
   363→}
