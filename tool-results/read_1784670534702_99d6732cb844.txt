   296→function linePath(points: Array<[number, number]>): string {
   297→  if (!points.length) return "";
   298→  return points.map(([x, y], i) => (i === 0 ? `M${x},${y}` : `L${x},${y}`)).join(" ");
   299→}
   300→
   301→function areaPath(points: Array<[number, number]>, baseY: number): string {
   302→  if (!points.length) return "";
   303→  const top = points.map(([x, y], i) => (i === 0 ? `M${x},${y}` : `L${x},${y}`)).join(" ");
   304→  const last = points[points.length - 1];
   305→  const first = points[0];
   306→  return `${top} L${last[0]},${baseY} L${first[0]},${baseY} Z`;
   307→}

   351→export function ReviewsView() {
   352→  const [location, setLocation] = React.useState<LocationKey>("todos");
   353→  const [ratingFilter, setRatingFilter] = React.useState<RatingFilter>("todos");
   354→  const [search, setSearch] = React.useState("");
   355→  const [selectedId, setSelectedId] = React.useState<string>(REVIEWS[0].id);
   356→  const [locationOpen, setLocationOpen] = React.useState(false);
   357→
   358→  const filtered = React.useMemo(() => {
   359→    return REVIEWS.filter((r) => {
   360→      if (location !== "todos") {
   361→        const locMap: Record<Exclude<LocationKey, "todos">, string> = {
   362→          madrid: "Ramses Madrid",
   363→          barcelona: "Ramses Barcelona",
   364→          valencia: "Ramses Valencia",
   365→        };
   366→        if (r.location !== locMap[location]) return false;
   367→      }
   368→      if (ratingFilter !== "todos" && r.rating !== ratingFilter) return false;
   369→      if (search.trim()) {
   370→        const q = search.toLowerCase();
   371→        if (!r.author.toLowerCase().includes(q) && !r.snippet.toLowerCase().includes(q) && !r.fullText.toLowerCase().includes(q)) {
   372→          return false;
   373→        }
   374→      }
   375→      return true;
   376→    });
   377→  }, [location, ratingFilter, search]);
   378→
   379→  const selected = React.useMemo(
   380→    () => REVIEWS.find((r) => r.id === selectedId) ?? REVIEWS[0],
   381→    [selectedId]
   382→  );
   383→
   384→  const locationLabel = LOCATIONS.find((l) => l.id === location)?.label ?? "Todos";
   385→
   386→  return (
   387→    <div className="space-y-5">
   388→      <div className="flex items-center justify-between gap-3 flex-wrap">
   389→        <div>
   390→          <h1 className="font-display text-2xl sm:text-3xl font-light tracking-tight">
   391→            Google Reviews
   392→          </h1>
   393→          <p className="text-sm text-muted-foreground mt-1">
   394→            Reputación online, análisis de sentimiento y respuestas asistidas por IA.
   395→          </p>
   396→        </div>
   397→        <DemoBadge />
   398→      </div>
   399→
   400→      <HeaderSummary
   401→        locationLabel={locationLabel}
   402→        locationOpen={locationOpen}
   403→        onToggleLocation={() => setLocationOpen((v) => !v)}
   404→        onSelectLocation={(l) => {
   405→          setLocation(l);
   406→          setLocationOpen(false);
   407→        }}
   408→        ratingFilter={ratingFilter}
   409→        onRatingFilter={setRatingFilter}
   410→        search={search}
   411→        onSearch={setSearch}
   412→      />
   413→
   414→      <div className="grid lg:grid-cols-[minmax(0,1fr)_minmax(0,1.35fr)] gap-4">
   415→        <ReviewsList reviews={filtered} selectedId={selected.id} onSelect={setSelectedId} />
   416→        <ReviewDetail review={selected} />
   417→      </div>
   418→
   419→      <StarEvolutionChart />
   420→
   421→      <CopilotPanel />
   422→    </div>
   423→  );
   424→}
