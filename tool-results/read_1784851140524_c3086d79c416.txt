   293→/* =========================================================
   294→ * Command palette (⌘K)
   295→ * Quick actions modal with search and keyboard navigation.
   296→ * Navigates via useNav.getState().go(section).
   297→ * =======================================================*/
   298→interface CmdAction {
   299→  id: string;
   300→  label: string;
   301→  hint: string;
   302→  icon: React.ElementType;
   303→  group: "Navegación" | "Acciones";
   304→  run: () => void;
   305→}
   306→
   307→function CommandPalette({ open, onOpenChange }: { open: boolean; onOpenChange: (o: boolean) => void }) {
   308→  const [query, setQuery] = React.useState("");
   309→  const [active, setActive] = React.useState(0);
   310→  const inputRef = React.useRef<HTMLInputElement>(null);
   311→  const listRef = React.useRef<HTMLDivElement>(null);
   312→
   313→  const actions = React.useMemo<CmdAction[]>(() => {
   314→    const nav: CmdAction[] = NAV.map((n) => ({
   315→      id: `nav-${n.id}`,
   316→      label: n.label,
   317→      hint: "Ir a sección",
   318→      icon: n.icon,
   319→      group: "Navegación",
   320→      run: () => useNav.getState().go(n.id),
   321→    }));
   322→    const quick: CmdAction[] = [
   323→      { id: "qa-executive", label: "Abrir Centro Ejecutivo", hint: "IA + Alertas", icon: Crown, group: "Acciones", run: () => useNav.getState().go("executive") },
   324→      { id: "qa-new-reserva", label: "Nueva reserva", hint: "Crear reserva", icon: CalendarPlus, group: "Acciones", run: () => useNav.getState().go("reservas") },
   325→      { id: "qa-search-client", label: "Buscar cliente", hint: "Abrir CRM", icon: UserSearch, group: "Acciones", run: () => useNav.getState().go("crm") },
   326→      { id: "qa-floor-plan", label: "Ver plano de mesas", hint: "Abrir reservas", icon: MapIcon, group: "Acciones", run: () => useNav.getState().go("reservas") },
   327→      { id: "qa-marketing", label: "Lanzar campaña", hint: "Marketing", icon: Megaphone, group: "Acciones", run: () => useNav.getState().go("marketing") },
   328→      { id: "qa-growth-analytics", label: "Ver Growth Analytics", hint: "ROI e insights IA", icon: TrendingUp, group: "Acciones", run: () => useNav.getState().go("growth-analytics") },
   329→      { id: "qa-growth-reputation", label: "Gestionar reputación", hint: "Reseñas + NPS", icon: Award, group: "Acciones", run: () => useNav.getState().go("growth-reputation") },
   330→      { id: "qa-reviews", label: "Responder reviews", hint: "Google Reviews", icon: Star, group: "Acciones", run: () => useNav.getState().go("reviews") },
   331→      { id: "qa-analytics", label: "Ver analítica", hint: "Analytics", icon: BarChart3, group: "Acciones", run: () => useNav.getState().go("analytics") },
   332→      { id: "qa-settings", label: "Configuración", hint: "Ajustes del local", icon: Settings, group: "Acciones", run: () => useNav.getState().go("configuracion") },
   333→    ];
   334→    return [...quick, ...nav];
   335→  }, []);
   336→
   337→  const filtered = React.useMemo(() => {
   338→    const q = query.trim().toLowerCase();
   339→    if (!q) return actions;
   340→    return actions.filter((a) =>
   341→      a.label.toLowerCase().includes(q) || a.hint.toLowerCase().includes(q)
   342→    );
   343→  }, [actions, query]);
   344→
   345→  // Reset state on open
   346→  React.useEffect(() => {
   347→    if (open) {
   348→      setQuery("");
   349→      setActive(0);
   350→      // focus input after dialog mounts
   351→      requestAnimationFrame(() => inputRef.current?.focus());
   352→    }
   353→  }, [open]);
   354→
   355→  // Keep active index in range
   356→  React.useEffect(() => {
   357→    if (active >= filtered.length) setActive(0);
   358→  }, [filtered, active]);
   359→
   360→  // Scroll active item into view
   361→  React.useEffect(() => {
   362→    const el = listRef.current?.querySelector<HTMLElement>(`[data-idx="${active}"]`);
   363→    el?.scrollIntoView({ block: "nearest" });
   364→  }, [active]);
   365→
   366→  function execute(a: CmdAction) {
   367→    a.run();
   368→    onOpenChange(false);
   369→  }
   370→
   371→  function onKeyDown(e: React.KeyboardEvent) {
   372→    if (e.key === "ArrowDown") {
   373→      e.preventDefault();
   374→      setActive((i) => (i + 1) % Math.max(filtered.length, 1));
   375→    } else if (e.key === "ArrowUp") {
   376→      e.preventDefault();
   377→      setActive((i) => (i - 1 + filtered.length) % Math.max(filtered.length, 1));
   378→    } else if (e.key === "Enter") {
   379→      e.preventDefault();
   380→      const a = filtered[active];
   381→      if (a) execute(a);
   382→    } else if (e.key === "Escape") {
   383→      onOpenChange(false);
   384→    }
   385→  }
   386→
   387→  // Group filtered actions for rendering
   388→  const groups = React.useMemo(() => {
   389→    const map = new Map<string, CmdAction[]>();
   390→    for (const a of filtered) {
   391→      const arr = map.get(a.group) ?? [];
   392→      arr.push(a);
   393→      map.set(a.group, arr);
   394→    }
   395→    return Array.from(map.entries());
   396→  }, [filtered]);
   397→
   398→  // Flat index lookup for keyboard nav
   399→  let flatIdx = -1;
   400→
   401→  return (
   402→    <Dialog open={open} onOpenChange={onOpenChange}>
   403→      <DialogContent
   404→        showCloseButton={false}
   405→        className="p-0 gap-0 sm:max-w-lg max-h-[80vh] overflow-hidden flex flex-col"
   406→        onKeyDown={onKeyDown}
   407→      >
   408→        <DialogTitle className="sr-only">Paleta de comandos</DialogTitle>
   409→        <DialogDescription className="sr-only">
   410→          Busca acciones rápidas o secciones y navega con el teclado.
   411→        </DialogDescription>
   412→        {/* Search input */}
   413→        <div className="flex items-center gap-2.5 px-4 h-14 border-b border-border/60">
   414→          <Search className="h-4 w-4 text-muted-foreground shrink-0" aria-hidden />
   415→          <input
   416→            ref={inputRef}
   417→            type="search"
   418→            value={query}
   419→            onChange={(e) => { setQuery(e.target.value); setActive(0); }}
   420→            placeholder="Busca una acción o sección…"
   421→            className="flex-1 bg-transparent outline-none text-sm placeholder:text-muted-foreground min-w-0"
   422→            aria-label="Buscar acción"
   423→            autoComplete="off"
   424→            spellCheck={false}
   425→          />
