   397→/** Animated number counter — uses requestAnimationFrame, transform/opacity only. */
   398→function useAnimatedNumber(value: number, duration = 350) {
   399→  const reduce = useReducedMotion();
   400→  const [display, setDisplay] = React.useState(value);
   401→  const fromRef = React.useRef(value);
   402→  const rafRef = React.useRef<number | null>(null);
   403→
   404→  React.useEffect(() => {
   405→    if (reduce) {
   406→      setDisplay(value);
   407→      return;
   408→    }
   409→    const from = fromRef.current;
   410→    const to = value;
   411→    if (from === to) return;
   412→    const start = performance.now();
   413→    function tick(now: number) {
   414→      const t = Math.min(1, (now - start) / duration);
   415→      // easeOutCubic
   416→      const eased = 1 - Math.pow(1 - t, 3);
   417→      setDisplay(Math.round(from + (to - from) * eased));
   418→      if (t < 1) {
   419→        rafRef.current = requestAnimationFrame(tick);
   420→      } else {
   421→        fromRef.current = to;
   422→      }
   423→    }
   424→    rafRef.current = requestAnimationFrame(tick);
   425→    return () => {
   426→      if (rafRef.current) cancelAnimationFrame(rafRef.current);
   427→      fromRef.current = value;
   428→    };
   429→  }, [value, duration, reduce]);
   430→
   431→  return display;
   432→}

   743→  function onCanvasDrop(e: React.DragEvent) {
   744→    if (!editMode || !draggedId) return;
   745→    e.preventDefault();
   746→    const rect = canvasRef.current?.getBoundingClientRect();
   747→    if (!rect) return;
   748→    const t = tables.find((x) => x.id === draggedId);
   749→    if (!t) return;
   750→    const w = tableW(t);
   751→    const h = tableH(t);
   752→    const x = Math.max(
   753→      4,
   754→      Math.min(e.clientX - rect.left - w / 2, rect.width - w - 4)
   755→    );
   756→    const y = Math.max(
   757→      4,
   758→      Math.min(e.clientY - rect.top - h / 2, rect.height - h - 4)
   759→    );
   760→    const nx = Math.round(x);
   761→    const ny = Math.round(y);
   762→    setTables((ts) =>
   763→      ts.map((x2) => (x2.id === draggedId ? { ...x2, x: nx, y: ny } : x2))
   764→    );
   765→    setDragOver(false);
   766→    setDraggedId(null);
   767→    toast({
   768→      title: `${t.name} reposicionada`,
   769→      description: `x ${nx} · y ${ny}`,
   770→    });
   771→  }

   779→  function onTableTouchStart(e: React.TouchEvent, t: RpTable) {
   780→    if (!editMode) return;
   781→    if (e.touches.length !== 1) return;
   782→    const rect = canvasRef.current?.getBoundingClientRect();
   783→    if (!rect) return;
   784→    const touch = e.touches[0];
   785→    touchDragRef.current = {
   786→      id: t.id,
   787→      offsetX: touch.clientX - rect.left - t.x,
   788→      offsetY: touch.clientY - rect.top - t.y,
   789→    };
   790→    setDraggedId(t.id);
   791→    setDragOver(true);
   792→  }

   794→  function onCanvasTouchMove(e: React.TouchEvent) {
   795→    if (!editMode || !touchDragRef.current) return;
   796→    if (e.touches.length !== 1) return;
   797→    e.preventDefault(); // stop page scroll while dragging
   798→    const rect = canvasRef.current?.getBoundingClientRect();
   799→    if (!rect) return;
   800→    const touch = e.touches[0];
   801→    const drag = touchDragRef.current;
   802→    const t = tables.find((x) => x.id === drag.id);
   803→    if (!t) return;
   804→    const w = tableW(t);
   805→    const h = tableH(t);
   806→    const x = Math.max(
   807→      4,
   808→      Math.min(touch.clientX - rect.left - drag.offsetX, rect.width - w - 4)
   809→    );
   810→    const y = Math.max(
   811→      4,
   812→      Math.min(touch.clientY - rect.top - drag.offsetY, rect.height - h - 4)
   813→    );
   814→    setTables((ts) =>
   815→      ts.map((x2) => (x2.id === drag.id ? { ...x2, x: Math.round(x), y: Math.round(y) } : x2))
   816→    );
   817→  }

   819→  function onCanvasTouchEnd() {
   820→    if (!touchDragRef.current) return;

   822→    touchDragRef.current = null;
   823→    setDraggedId(null);
   824→    setDragOver(false);
   825→    if (t) {
   826→      toast({
   827→        title: `${t.name} reposicionada`,
   828→        description: `x ${t.x} · y ${t.y}`,
   829→      });
   830→    }
