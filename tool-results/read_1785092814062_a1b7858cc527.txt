   487→/* ---------- Sparkline ---------- */
   488→
   489→function Sparkline({ data, color }: { data: number[]; color: "gold" | "teal" }) {
   490→  const rawId = React.useId();
   491→  const id = rawId.replace(/[:]/g, "");
   492→  const w = 76;
   493→  const h = 28;
   494→  const pad = 3;
   495→  const max = Math.max(...data);
   496→  const min = Math.min(...data);
   497→  const range = max - min || 1;
   498→  const pts = data.map((v, i) => {
   499→    const x = (i / (data.length - 1)) * (w - pad * 2) + pad;
   500→    const y = h - pad - ((v - min) / range) * (h - pad * 2);
   501→    return [x, y] as [number, number];
   502→  });
   503→  const linePts = pts.map(([x, y]) => `${x.toFixed(1)},${y.toFixed(1)}`).join(" ");
   504→  const areaPath = `M${pts[0][0].toFixed(1)},${(h - pad).toFixed(1)} L${pts
   505→    .map(([x, y]) => `${x.toFixed(1)},${y.toFixed(1)}`)
   506→    .join(" L")} L${pts[pts.length - 1][0].toFixed(1)},${(h - pad).toFixed(1)} Z`;
   507→  const stroke = color === "gold" ? "var(--gold)" : "var(--teal)";
   508→  const lastPt = pts[pts.length - 1];
   509→
   510→  return (
   511→    <svg
   512→      width={w}
   513→      height={h}
   514→      viewBox={`0 0 ${w} ${h}`}
   515→      aria-hidden="true"
   516→      focusable="false"
   517→      className="shrink-0 overflow-visible max-w-full h-auto"
   518→      preserveAspectRatio="xMidYMid meet"
   519→    >
   520→      <defs>
   521→        <linearGradient id={`spark-${id}`} x1="0" x2="0" y1="0" y2="1">
   522→          <stop offset="0%" stopColor={stroke} stopOpacity="0.32" />
   523→          <stop offset="100%" stopColor={stroke} stopOpacity="0" />
   524→        </linearGradient>
   525→      </defs>
   526→      <path d={areaPath} fill={`url(#spark-${id})`} />
   527→      <polyline
   528→        points={linePts}
   529→        fill="none"
   530→        stroke={stroke}
   531→        strokeWidth="1.5"
   532→        strokeLinejoin="round"
   533→        strokeLinecap="round"
   534→      />
   535→      <circle cx={lastPt[0]} cy={lastPt[1]} r="2" fill={stroke} />
   536→    </svg>
   537→  );
   538→}

   547→  const pillCls = isGood
