    85→function useReducedMotion() {
    86→  const [reduced, setReduced] = React.useState(false);
    87→  React.useEffect(() => {
    88→    const mq = window.matchMedia("(prefers-reduced-motion: reduce)");
    89→    const update = () => setReduced(mq.matches);
    90→    update();
    91→    mq.addEventListener?.("change", update);
    92→    return () => mq.removeEventListener?.("change", update);
    93→  }, []);
    94→  return reduced;
    95→}

    97→function useInView<T extends HTMLElement>(opts?: IntersectionObserverInit) {
    98→  const ref = React.useRef<T | null>(null);
    99→  const [inView, setInView] = React.useState(false);
   100→  React.useEffect(() => {
   101→    const el = ref.current;
   102→    if (!el) return;
   103→    const ob = new IntersectionObserver(
   104→      ([entry]) => {
   105→        if (entry.isIntersecting) {
   106→          setInView(true);
   107→          ob.disconnect();
   108→        }
   109→      },
   110→      opts ?? { threshold: 0.2 },
   111→    );
   112→    ob.observe(el);
   113→    return () => ob.disconnect();
   114→  }, [opts]);
   115→  return { ref, inView };
   116→}

   118→function useCountUp(target: number, run: boolean, duration = 1400) {
   119→  const reduced = useReducedMotion();
   120→  const [val, setVal] = React.useState(0);
   121→  React.useEffect(() => {
   122→    if (!run) return;
   123→    if (reduced) {
   124→      setVal(target);
   125→      return;
   126→    }
   127→    let raf = 0;
   128→    const start = performance.now();
   129→    const tick = (now: number) => {
   130→      const t = Math.min(1, (now - start) / duration);
   131→      const eased = 1 - Math.pow(1 - t, 3);
   132→      setVal(target * eased);
   133→      if (t < 1) raf = requestAnimationFrame(tick);
   134→      else setVal(target);
   135→    };
   136→    raf = requestAnimationFrame(tick);
   137→    return () => cancelAnimationFrame(raf);
   138→  }, [target, run, reduced, duration]);
   139→  return val;
   140→}
