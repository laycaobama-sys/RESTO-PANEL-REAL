   195→function fmtLimit(v: number | null | boolean, unit?: string): string {
   196→  if (v === true) return "✔";
   197→  if (v === false) return "✖";

   199→  if (v === null) return "∞";

   238→function useUsageState(current: number, limit: number): {
   239→  pct: number;
   240→  tone: "ok" | "warn" | "danger" | "blocked" | "unlimited";
   241→} {
   242→  if (limit === -1) return { pct: 0, tone: "unlimited" };
   243→  if (limit === 0) return { pct: 0, tone: "blocked" };
   244→  const pct = Math.min(100, Math.round((current / limit) * 100));
   245→  let tone: "ok" | "warn" | "danger" = "ok";
   246→  if (pct >= 100) tone = "danger";
   247→  else if (pct >= 80) tone = "warn";
   248→  return { pct, tone };
   249→}

   251→/** Pure entitlements engine — can(org, feature). */
   252→function can(plan: PlanId, feature: string): boolean {
   253→  const row = FEATURES.find((f) => f.feature === feature);
   254→  if (!row) return false;
   255→  const v = row.values[plan];
   256→  if (v === true) return true;
   257→  if (v === false || v === 0) return false;
   258→  return true; // numeric (incl. -1) = available
   259→}
   260→
   261→/** limit(org, feature) → numeric limit, -1 = ∞, 0 = blocked. */
   262→function limitOf(plan: PlanId, feature: string): number {
   263→  const row = FEATURES.find((f) => f.feature === feature);
   264→  if (!row) return 0;
   265→  const v = row.values[plan];
   266→  if (v === true) return -1;
   267→  if (v === false) return 0;
   268→  return v;
   269→}
