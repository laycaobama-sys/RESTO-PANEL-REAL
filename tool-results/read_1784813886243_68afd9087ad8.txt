   398→function roiColor(roi: number): string {
   399→  if (roi >= 200) return "text-emerald-300";
   400→  if (roi >= 100) return "text-[var(--gold-soft)]";
   401→  return "text-rose-300";
   402→}
   403→
   404→function roiOf(c: Campaign): number {
   405→  if (c.cost === 0) return c.revenue > 0 ? 999 : 0;
   406→  return Math.round(((c.revenue - c.cost) / c.cost) * 100);
   407→}
   408→
   409→function pct(n: number, d: number): string {
   410→  if (d === 0) return "—";
   411→  return `${((n / d) * 100).toFixed(0)}%`;
   412→}

   414→function fmtEUR(n: number): string {
   415→  return n.toLocaleString("es-ES", { style: "currency", currency: "EUR", minimumFractionDigits: 0, maximumFractionDigits: 0 });
   416→}
