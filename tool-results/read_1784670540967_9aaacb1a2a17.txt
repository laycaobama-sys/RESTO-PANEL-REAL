   483→   483→function nodeSummary(node: FlowNode): string {
   484→   484→  switch (node.type) {
   485→   485→    case "trigger": {
   486→   486→      const e = TRIGGER_EVENTS.find((t) => t.value === node.config.event);
   487→   487→      return `Cuando: ${e?.label ?? node.config.event ?? "—"}`;
   488→   488→    }
   489→   489→    case "condition":
   490→   490→      return `Si: ${node.config.field ?? "—"} ${node.config.operator ?? "=="} ${node.config.value ?? "—"}`;
   491→   491→    case "action": {
   492→   492→      const a = ACTION_META[node.config.actionType ?? "send_whatsapp"];
   493→   493→      return `Entonces: ${a.verb(node.config)}`;
   494→   494→    }
   495→   495→    case "wait": {
   496→   496→      const u = WAIT_UNITS.find((x) => x.value === node.config.unit);
   497→   497→      const anchor = WAIT_ANCHORS.find((x) => x.value === node.config.anchor);
   498→   498→      return `Esperar: ${node.config.duration ?? 0}${u?.short ?? "h"}${anchor ? ` · ${anchor.label}` : ""}`;
   499→   499→    }
   500→   500→    case "branch":
   501→   501→      return `Si: ${node.config.ifLabel ?? "—"} · Si no: ${node.config.elseLabel ?? "—"}`;
   502→   502→    default:
   503→   503→      return "";
   504→   504→  }
   505→   505→}

   511→   511→      return `plantilla: ${node.config.template}`;

   514→   514→      return `valor: ${node.config.value ?? "—"}`;

   516→   516→    if (at === "notify_staff") return `canal: ${node.config.channel ?? "—"}`;

   522→   522→function buildSimLog(nodes: FlowNode[], active: boolean, flowName: string): { runId: string; lines: SimLine[] } {
   523→   523→  const lines: SimLine[] = [];
   524→   524→  const start = new Date();
   525→   525→  const at = (sec: number) => {
   526→   526→    const d = new Date(start.getTime() + sec * 1000);
   527→   527→    return d.toTimeString().slice(0, 8);
   528→   528→  };
   529→   529→  const runId = `RUN-${2000 + Math.floor(Math.random() * 7000)}`;
   530→   530→  lines.push({ ts: at(0), level: "info", msg: `Ejecución iniciada — flujo "${flowName}" (#${runId})` });
   531→   531→  lines.push({ ts: at(0), level: "info", msg: `Modo: dry-run · ${nodes.length} pasos en cola · datos demo` });
   532→   532→  if (!active) {
   533→   533→    lines.push({ ts: at(0), level: "warn", msg: "El flujo está pausado en producción — la simulación recorre los pasos igualmente." });
   534→   534→  }
   535→   535→  let t = 0.2;
   536→   536→  for (const n of nodes) {
   537→   537→    t += 0.05;
   538→   538→    const ts = at(Math.floor(t));
   539→   539→    switch (n.type) {
   540→   540→      case "trigger": {
   541→   541→        const e = TRIGGER_EVENTS.find((x) => x.value === n.config.event);
   542→   542→        lines.push({ ts, level: "info", msg: `▸ Trigger: ${e?.label ?? n.config.event} — coincidencia encontrada (RES-9182, mesa 14, 4 pax)` });
   543→   543→        lines.push({ ts, level: "info", msg: `  Cliente resuelto: Marta Ruiz (cust_8821, +34 612 88 21 04)` });
   544→   544→        break;
   545→   545→      }
   546→   546→      case "wait": {
   547→   547→        const u = WAIT_UNITS.find((x) => x.value === n.config.unit);
   548→   548→        const anchor = WAIT_ANCHORS.find((x) => x.value === n.config.anchor);
   549→   549→        lines.push({ ts, level: "info", msg: `▸ Wait: programando ${n.config.duration ?? 0}${u?.short ?? "h"} ${anchor?.label ?? ""}` });
   550→   550→        lines.push({ ts, level: "info", msg: `  Ejecución pausada — reanudará al alcanzar el ancla` });
   551→   551→        t += 1;
   552→   552→        lines.push({ ts: at(Math.floor(t)), level: "info", msg: `  ── salto simulado al instante de reanudación ──` });
   553→   553→        break;
   554→   554→      }
   555→   555→      case "condition": {
   556→   556→        const v = n.config.value ?? "";
   557→   557→        const result = v === "true" || v === "4" || v === "60" || v === "2" ? true : Math.floor(t * 13) % 2 === 0;
   558→   558→        lines.push({
   559→   559→          ts,
   560→   560→          level: result ? "success" : "warn",
   561→   561→          msg: `▸ Condition: ${n.config.field} ${n.config.operator} ${n.config.value} → ${result ? "TRUE" : "FALSE"}`,
   562→   562→        });
   563→   563→        if (!result) {
   564→   564→          lines.push({ ts, level: "warn", msg: "  Condición no cumplida — el flujo continuaría por la rama alternativa (si existe)." });
   565→   565→        }
   566→   566→        break;
   567→   567→      }
   568→   568→      case "action": {
   569→   569→        const a = ACTION_META[n.config.actionType ?? "send_whatsapp"];
   570→   570→        const tpl = n.config.template ? ` · plantilla "${n.config.template}"` : "";
   571→   571→        lines.push({ ts, level: "info", msg: `▸ Action: ${a.label}${tpl}` });
   572→   572→        lines.push({ ts, level: "success", msg: `  ${a.successMsg} (latencia ${Math.floor(80 + Math.random() * 140)}ms)` });
   573→   573→        break;
   574→   574→      }
   575→   575→      case "branch": {
   576→   576→        lines.push({ ts, level: "info", msg: `▸ Branch: evaluando condición interna — rama "${n.config.ifLabel}" seleccionada` });
   577→   577→        break;
   578→   578→      }
   579→   579→    }
   580→   580→    t += 0.1;
   581→   581→  }
   582→   582→  const total = t.toFixed(2);
   583→   583→  lines.push({ ts: at(Math.ceil(t)), level: "success", msg: `Ejecución finalizada — estado: success · duración total: ${total}s` });
   584→   584→  return { runId, lines };
   585→   585→}
