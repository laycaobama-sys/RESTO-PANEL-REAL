   100→/* ---------------- permissions matrix ---------------- */
   101→function PermissionsMatrix({
   102→  perms, onChange, readOnly,
   103→}: {
   104→  perms: Record<string, boolean[]>;
   105→  onChange?: (resource: string, actionIdx: number, value: boolean) => void;
   106→  readOnly?: boolean;
   107→}) {
   108→  return (
   109→    <div className="overflow-x-auto rp-scroll-thin rounded-lg border border-border/40">
   110→      <table className="w-full border-collapse text-xs">
   111→        <thead>
   112→          <tr className="border-b border-border/60 bg-foreground/[0.03]">
   113→            <th className="px-3 py-2 text-left text-[10px] font-mono uppercase tracking-wider text-muted-foreground sticky left-0 bg-background/95">Recurso</th>
   114→            {ACTIONS.map((a) => (
   115→              <th key={a} className="px-3 py-2 text-center text-[10px] font-mono uppercase tracking-wider text-muted-foreground capitalize">{a}</th>
   116→            ))}
   117→          </tr>
   118→        </thead>
   119→        <tbody>
   120→          {RESOURCES.map((r) => (
   121→            <tr key={r} className="border-b border-border/40 last:border-0">
   122→              <td className="px-3 py-2 font-medium sticky left-0 bg-background/95">{r}</td>
   123→              {ACTIONS.map((a, ai) => {
   124→                const checked = perms[r]?.[ai] ?? false;
   125→                return (
   126→                  <td key={a} className="px-3 py-2 text-center">
   127→                    {readOnly ? (
   128→                      checked ? <Check className="h-3.5 w-3.5 text-emerald-400 inline" aria-label="Permitido" />
   129→                             : <X className="h-3.5 w-3.5 text-muted-foreground/40 inline" aria-label="Denegado" />
   130→                    ) : (
   131→                      <Checkbox checked={checked} onCheckedChange={(v) => onChange?.(r, ai, !!v)} aria-label={`${a} ${r}`} />
   132→                    )}
   133→                  </td>
   134→                );
   135→              })}
   136→            </tr>
   137→          ))}
   138→        </tbody>
   139→      </table>
   140→    </div>
   141→  );
   142→}

   152→  const [removeTarget, setRemoveTarget] = React.useState<Member | null>(null);
   153→
   154→  // invite form
   155→  const [invite, setInvite] = React.useState({ email: "", role: "Gerente", locations: [] as string[] });
   156→  const [inviteErr, setInviteErr] = React.useState<string | null>(null);
   157→
   158→  // custom role form
   159→  const [customRole, setCustomRole] = React.useState({ name: "", perms: Object.fromEntries(RESOURCES.map((r) => [r, [false, false, false, false]])) as Record<string, boolean[]> });
   160→  const [customErr, setCustomErr] = React.useState<string | null>(null);
   161→
   162→  // permissions editor (copy from defaults)
   163→  const [editPerms, setEditPerms] = React.useState<Record<string, boolean[]> | null>(null);
   164→
   165→  const filtered = MEMBERS.filter((m) => {
   166→    if (search && !`${m.name} ${m.email}`.toLowerCase().includes(search.toLowerCase())) return false;
   167→    if (roleFilter !== "Todos" && m.role !== roleFilter) return false;
   168→    return true;
   169→  });
   170→
   171→  const openPerms = (roleId: string) => {
   172→    const base = DEFAULT_PERMS[roleId] ?? Object.fromEntries(RESOURCES.map((r) => [r, [false, false, false, false]]));
   173→    setEditPerms(JSON.parse(JSON.stringify(base)));
   174→    setPermsOpen(roleId);
   175→  };
   176→
   177→  const savePerms = () => {
   178→    toast({ title: "Permisos actualizados (demo)", description: `Rol ${ROLES.find((r) => r.id === permsOpen)?.name} guardado.` });
   179→    setPermsOpen(null);
   180→  };
   181→
   182→  const togglePerm = (resource: string, actionIdx: number, value: boolean) => {
   183→    setEditPerms((p) => {
   184→      if (!p) return p;
   185→      const next = { ...p, [resource]: [...(p[resource] ?? [false, false, false, false])] };
   186→      next[resource][actionIdx] = value;
   187→      return next;
   188→    });
   189→  };
   190→
   191→  const submitInvite = () => {
   192→    setInviteErr(null);
   193→    if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(invite.email)) { setInviteErr("El email no es válido."); return; }
   194→    if (invite.locations.length === 0) { setInviteErr("Selecciona al menos un local."); return; }
   195→    toast({ title: "Invitación enviada (demo)", description: `${invite.email} · ${invite.role} · ${invite.locations.length} local(es)` });
   196→    setInvite({ email: "", role: "Gerente", locations: [] });
   197→    setInviteOpen(false);
   198→  };
   199→
   200→  const submitCustomRole = () => {
   201→    setCustomErr(null);
   202→    if (customRole.name.trim().length < 3) { setCustomErr("El nombre del rol debe tener al menos 3 caracteres."); return; }
   203→    const totalGranted = Object.values(customRole.perms).reduce((acc, arr) => acc + arr.filter(Boolean).length, 0);
   204→    if (totalGranted === 0) { setCustomErr("Concede al menos un permiso."); return; }
   205→    toast({ title: "Rol personalizado creado (demo)", description: `"${customRole.name}" con ${totalGranted} permisos.` });
   206→    setCustomRole({ name: "", perms: Object.fromEntries(RESOURCES.map((r) => [r, [false, false, false, false]])) });
   207→    setCustomRoleOpen(false);
   208→  };
   209→
   210→  const toggleCustomPerm = (resource: string, actionIdx: number, value: boolean) => {
   211→    setCustomRole((c) => {
   212→      const next = { ...c.perms, [resource]: [...(c.perms[resource] ?? [false, false, false, false])] };
   213→      next[resource][actionIdx] = value;
   214→      return { ...c, perms: next };
   215→    });
   216→  };
   217→
   218→  const confirmRemove = () => {
   219→    if (removeTarget) {
   220→      toast({ title: "Miembro eliminado (demo)", description: `${removeTarget.name} ya no tiene acceso.`, variant: "destructive" });
   221→      setRemoveTarget(null);
   222→    }
   223→  };
   224→
   225→  return (
   226→    <div className="space-y-6 max-w-[1400px] mx-auto">
   227→      <header className="space-y-2">
   228→        <div className="flex items-center gap-2.5">
   229→          <div className="h-9 w-9 rounded-md bg-[var(--teal)]/10 border border-[var(--teal)]/30 flex items-center justify-center">
   230→            <UserCog className="h-5 w-5 rp-teal-text" aria-hidden />
   231→          </div>
   232→          <h1 className="font-display text-2xl sm:text-3xl font-light tracking-tight">Equipo</h1>
   233→          <DemoBadge />
   234→        </div>
   235→        <p className="text-sm text-muted-foreground max-w-2xl">
   236→          Gestiona miembros, roles y permisos. La cuenta de Owner no puede ser eliminada ni degradada.
   237→        </p>
   238→      </header>
   239→
   240→      {/* KPIs */}
   241→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   242→        {[
   243→          { label: "Miembros", value: MEMBERS.length, sub: `${MEMBERS.filter((m) => m.status === "activa").length} activos` },
   244→          { label: "Roles disponibles", value: ROLES.length, sub: "11 sistema + custom" },
   245→          { label: "Invitaciones pendientes", value: MEMBERS.filter((m) => m.status === "invitada").length, sub: "enviadas" },
   246→          { label: "Suspendidos", value: MEMBERS.filter((m) => m.status === "suspendida").length, sub: "sin acceso" },
   247→        ].map((s) => (
   248→          <div key={s.label} className="rp-glass rounded-xl p-4">

   365→              <div className="flex items-start justify-between gap-2">
   366→                <div className="flex items-center gap-2">
   367→                  {r.id === "owner" ? <Crown className="h-4 w-4 rp-gold-text" aria-hidden /> : <Shield className="h-4 w-4 text-muted-foreground" aria-hidden />}

   379→                </Button>

   390→            <DialogTitle className="flex items-center gap-2"><UserPlus className="h-4 w-4" aria-hidden /> Invitar miembro <DemoBadge className="ml-1" /></DialogTitle>

   392→          </DialogHeader>

   394→            <div>

   397→                <Mail className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" aria-hidden />
   398→                <Input id="inv-email" type="email" placeholder="persona@empresa.com" value={invite.email} onChange={(e) => setInvite({ ...invite, email: e.target.value })} className="pl-9" />
   399→              </div>

   402→              <Label htmlFor="inv-role" className="text-xs">Rol *</Label>

   429→              <div className="rounded-md border border-rose-400/40 bg-rose-400/10 p-2.5 text-xs text-rose-200 flex items-start gap-2">
   430→                <X className="h-3.5 w-3.5 mt-0.5 shrink-0" aria-hidden /><span>{inviteErr}</span>
   431→              </div>
   432→            )}
   433→          </div>
   434→          <DialogFooter>
   435→            <Button variant="outline" onClick={() => setInviteOpen(false)}>Cancelar</Button>

   445→            <DialogTitle className="flex items-center gap-2">
   446→              <Shield className="h-4 w-4 rp-teal-text" aria-hidden />

   454→              <PermissionsMatrix perms={editPerms} onChange={togglePerm} />

   470→        <DialogContent className="sm:max-w-2xl">
   471→          <DialogHeader>
   472→            <DialogTitle className="flex items-center gap-2"><KeyRound className="h-4 w-4 rp-gold-text" aria-hidden /> Crear rol personalizado <DemoBadge className="ml-1" /></DialogTitle>
   473→            <DialogDescription>Define un rol a medida con permisos granulares.</DialogDescription>
   474→          </DialogHeader>
   475→          <div className="space-y-3">
   476→            <div>
   477→              <Label htmlFor="role-name" className="text-xs">Nombre del rol *</Label>
   478→              <Input id="role-name" placeholder="ej. Coordinador de eventos" value={customRole.name} onChange={(e) => setCustomRole({ ...customRole, name: e.target.value })} className="mt-1.5" />
   479→            </div>
   480→            <div>
   481→              <Label className="text-xs">Permisos</Label>
   482→              <div className="mt-1.5">
   483→                <PermissionsMatrix perms={customRole.perms} onChange={toggleCustomPerm} />
   484→              </div>
   485→            </div>
   486→            {customErr && (
   487→              <div className="rounded-md border border-rose-400/40 bg-rose-400/10 p-2.5 text-xs text-rose-200 flex items-start gap-2">
   488→                <X className="h-3.5 w-3.5 mt-0.5 shrink-0" aria-hidden /><span>{customErr}</span>
   489→              </div>
   490→            )}
   491→          </div>
   492→          <DialogFooter>
   493→            <Button variant="outline" onClick={() => setCustomRoleOpen(false)}>Cancelar</Button>
   494→            <Button onClick={submitCustomRole}>Crear rol</Button>
   495→          </DialogFooter>
   496→        </DialogContent>
   497→      </Dialog>
   498→
   499→      {/* Remove confirm */}
   500→      <AlertDialog open={!!removeTarget} onOpenChange={(o) => !o && setRemoveTarget(null)}>
   501→        <AlertDialogContent>
   502→          <AlertDialogHeader>

   504→            <AlertDialogDescription>

   506→              Esta acción revoca su acceso inmediatamente y no se puede deshacer.
   507→            </AlertDialogDescription>
   508→          </AlertDialogHeader>
   509→          <AlertDialogFooter>
   510→            <AlertDialogCancel>Cancelar</AlertDialogCancel>
   511→            <AlertDialogAction onClick={confirmRemove} className="bg-rose-500 hover:bg-rose-600 text-white">Sí, eliminar</AlertDialogAction>
