    80→function useReducedMotion() {
    81→  const [reduced, setReduced] = React.useState(false);
    82→  React.useEffect(() => {
    83→    const mq = window.matchMedia("(prefers-reduced-motion: reduce)");
    84→    const update = () => setReduced(mq.matches);
    85→    update();
    86→    mq.addEventListener?.("change", update);
    87→    return () => mq.removeEventListener?.("change", update);
    88→  }, []);
    89→  return reduced;
    90→}

    92→function useInView<T extends HTMLElement>(opts?: IntersectionObserverInit) {
    93→  const ref = React.useRef<T | null>(null);
    94→  const [inView, setInView] = React.useState(false);
    95→  React.useEffect(() => {
    96→    const el = ref.current;
    97→    if (!el) return;
    98→    const ob = new IntersectionObserver(
    99→      ([entry]) => {
   100→        if (entry.isIntersecting) {
   101→          setInView(true);
   102→          ob.disconnect();
   103→        }
   104→      },
   105→      opts ?? { threshold: 0.2 },
   106→    );
   107→    ob.observe(el);
   108→    return () => ob.disconnect();
   109→  }, [opts]);
   110→  return { ref, inView };
   111→}

   113→function useCountUp(target: number, run: boolean, duration = 1400) {
   114→  const reduced = useReducedMotion();
   115→  const [val, setVal] = React.useState(0);
   116→  React.useEffect(() => {
   117→    if (!run) return;
   118→    if (reduced) {
   119→      setVal(target);
   120→      return;
   121→    }
   122→    let raf = 0;
   123→    const start = performance.now();
   124→    const tick = (now: number) => {
   125→      const t = Math.min(1, (now - start) / duration);
   126→      const eased = 1 - Math.pow(1 - t, 3);
   127→      setVal(target * eased);
   128→      if (t < 1) raf = requestAnimationFrame(tick);
   129→      else setVal(target);
   130→    };
   131→    raf = requestAnimationFrame(tick);
   132→    return () => cancelAnimationFrame(raf);
   133→  }, [target, run, reduced, duration]);
   134→  return val;
   135→}
