   487→   487→/* ---------- Sparkline ---------- */
   488→   488→
   489→   489→function Sparkline({ data, color }: { data: number[]; color: "gold" | "teal" }) {
   490→   490→  const rawId = React.useId();
   491→   491→  const id = rawId.replace(/[:]/g, "");
   492→   492→  const w = 76;
   493→   493→  const h = 28;
   494→   494→  const pad = 3;
   495→   495→  const max = Math.max(...data);
   496→   496→  const min = Math.min(...data);
   497→   497→  const range = max - min || 1;
   498→   498→  const pts = data.map((v, i) => {
   499→   499→    const x = (i / (data.length - 1)) * (w - pad * 2) + pad;
   500→   500→    const y = h - pad - ((v - min) / range) * (h - pad * 2);
   501→   501→    return [x, y] as [number, number];
   502→   502→  });
   503→   503→  const linePts = pts.map(([x, y]) => `${x.toFixed(1)},${y.toFixed(1)}`).join(" ");
   504→   504→  const areaPath = `M${pts[0][0].toFixed(1)},${(h - pad).toFixed(1)} L${pts
   505→   505→    .map(([x, y]) => `${x.toFixed(1)},${y.toFixed(1)}`)
   506→   506→    .join(" L")} L${pts[pts.length - 1][0].toFixed(1)},${(h - pad).toFixed(1)} Z`;
   507→   507→  const stroke = color === "gold" ? "var(--gold)" : "var(--teal)";
   508→   508→  const lastPt = pts[pts.length - 1];
   509→   509→
   510→   510→  return (
   511→   511→    <svg
   512→   512→      width={w}
   513→   513→      height={h}
   514→   514→      viewBox={`0 0 ${w} ${h}`}
   515→   515→      aria-hidden="true"
   516→   516→      focusable="false"
   517→   517→      className="shrink-0 overflow-visible"
   518→   518→    >
   519→   519→      <defs>
   520→   520→        <linearGradient id={`spark-${id}`} x1="0" x2="0" y1="0" y2="1">
   521→   521→          <stop offset="0%" stopColor={stroke} stopOpacity="0.32" />
   522→   522→          <stop offset="100%" stopColor={stroke} stopOpacity="0" />
   523→   523→        </linearGradient>
   524→   524→      </defs>
   525→   525→      <path d={areaPath} fill={`url(#spark-${id})`} />
   526→   526→      <polyline
   527→   527→        points={linePts}
   528→   528→        fill="none"
   529→   529→        stroke={stroke}
   530→   530→        strokeWidth="1.5"
   531→   531→        strokeLinejoin="round"
   532→   532→        strokeLinecap="round"
   533→   533→      />
   534→   534→      <circle cx={lastPt[0]} cy={lastPt[1]} r="2" fill={stroke} />
   535→   535→    </svg>
   536→   536→  );
   537→   537→}
