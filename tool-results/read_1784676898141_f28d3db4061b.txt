    80→    80→function useReducedMotion() {
    81→    81→  const [reduced, setReduced] = React.useState(false);
    82→    82→  React.useEffect(() => {
    83→    83→    const mq = window.matchMedia("(prefers-reduced-motion: reduce)");
    84→    84→    const update = () => setReduced(mq.matches);
    85→    85→    update();
    86→    86→    mq.addEventListener?.("change", update);
    87→    87→    return () => mq.removeEventListener?.("change", update);
    88→    88→  }, []);
    89→    89→  return reduced;
    90→    90→}

    92→    92→function useInView<T extends HTMLElement>(opts?: IntersectionObserverInit) {
    93→    93→  const ref = React.useRef<T | null>(null);
    94→    94→  const [inView, setInView] = React.useState(false);
    95→    95→  React.useEffect(() => {
    96→    96→    const el = ref.current;
    97→    97→    if (!el) return;
    98→    98→    const ob = new IntersectionObserver(
    99→    99→      ([entry]) => {
   100→   100→        if (entry.isIntersecting) {
   101→   101→          setInView(true);
   102→   102→          ob.disconnect();
   103→   103→        }
   104→   104→      },
   105→   105→      opts ?? { threshold: 0.2 },
   106→   106→    );
   107→   107→    ob.observe(el);
   108→   108→    return () => ob.disconnect();
   109→   109→  }, [opts]);
   110→   110→  return { ref, inView };
   111→   111→}

   113→   113→function useCountUp(target: number, run: boolean, duration = 1400) {
   114→   114→  const reduced = useReducedMotion();
   115→   115→  const [val, setVal] = React.useState(0);
   116→   116→  React.useEffect(() => {
   117→   117→    if (!run) return;
   118→   118→    if (reduced) {
   119→   119→      setVal(target);
   120→   120→      return;
   121→   121→    }
   122→   122→    let raf = 0;
   123→   123→    const start = performance.now();
   124→   124→    const tick = (now: number) => {
   125→   125→      const t = Math.min(1, (now - start) / duration);
   126→   126→      const eased = 1 - Math.pow(1 - t, 3);
   127→   127→      setVal(target * eased);
   128→   128→      if (t < 1) raf = requestAnimationFrame(tick);
   129→   129→      else setVal(target);
   130→   130→    };
   131→   131→    raf = requestAnimationFrame(tick);
   132→   132→    return () => cancelAnimationFrame(raf);
   133→   133→  }, [target, run, reduced, duration]);
   134→   134→  return val;
   135→   135→}
