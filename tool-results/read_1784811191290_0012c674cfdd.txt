   117→function heatFor(dayIdx: number, hour: number): number {
   118→  const dayBoost = [0.7, 0.75, 0.85, 0.95, 1.1, 1.25, 1.0][dayIdx];
   119→  let base = 0;
   120→  if (hour < 10) base = 0.05;
   121→  else if (hour < 12) base = 0.2 + (hour - 10) * 0.15;
   122→  else if (hour === 12) base = 0.6;
   123→  else if (hour <= 15) base = 0.85 - (hour - 13) * 0.1;
   124→  else if (hour <= 18) base = 0.35 + (hour - 16) * 0.05;
   125→  else if (hour === 19) base = 0.6;
   126→  else if (hour === 20) base = 0.85;
   127→  else if (hour === 21) base = 0.92;
   128→  else if (hour === 22) base = 0.7;
   129→  else base = 0.2;
   130→  return Math.min(1, base * dayBoost);
   131→}

   176→function linePath(points: Array<[number, number]>): string {
   177→  if (!points.length) return "";
   178→  return points.map(([x, y], i) => (i === 0 ? `M${x},${y}` : `L${x},${y}`)).join(" ");
   179→}
   180→
   181→function areaPath(points: Array<[number, number]>, baseY: number): string {
   182→  if (!points.length) return "";
   183→  const top = points.map(([x, y], i) => (i === 0 ? `M${x},${y}` : `L${x},${y}`)).join(" ");
   184→  const last = points[points.length - 1];
   185→  const first = points[0];
   186→  return `${top} L${last[0]},${baseY} L${first[0]},${baseY} Z`;
   187→}
   188→
   189→function bandPath(
   190→  upper: Array<[number, number]>,
   191→  lower: Array<[number, number]>
   192→): string {
   193→  if (!upper.length || !lower.length) return "";
   194→  const top = upper.map(([x, y], i) => (i === 0 ? `M${x},${y}` : `L${x},${y}`)).join(" ");
   195→  const bottom = lower
   196→    .slice()
   197→    .reverse()
   198→    .map(([x, y]) => `L${x},${y}`)
   199→    .join(" ");
   200→  return `${top} ${bottom} Z`;
   201→}

   351→function Sparkline({ values, color }: { values: number[]; color: string }) {
   352→  const W = 80;
   353→  const H = 24;
   354→  const pad = 2;
   355→  const min = Math.min(...values);
   356→  const max = Math.max(...values);
   357→  const range = max - min || 1;
   358→  const pts: Array<[number, number]> = values.map((v, i) => {
   359→    const x = pad + (i / (values.length - 1)) * (W - 2 * pad);
   360→    const y = pad + (1 - (v - min) / range) * (H - 2 * pad);
   361→    return [x, y];
   362→  });
   363→  const baseY = H - pad;
   364→  return (
   365→    <svg
   366→      viewBox={`0 0 ${W} ${H}`}
   367→      className="w-20 h-6"
   368→      role="img"
   369→      aria-label={`Tendencia ${values[0]} a ${values[values.length - 1]}`}
   370→    >
   371→      <defs>
   372→        <linearGradient id={`spark-${color.replace(/[^a-z0-9]/gi, "")}`} x1="0" y1="0" x2="0" y2="1">
   373→          <stop offset="0%" stopColor={color} stopOpacity="0.4" />
   374→          <stop offset="100%" stopColor={color} stopOpacity="0" />
   375→        </linearGradient>
   376→      </defs>
   377→      <path d={areaPath(pts, baseY)} fill={`url(#spark-${color.replace(/[^a-z0-9]/gi, "")})`} />
   378→      <path d={linePath(pts)} fill="none" stroke={color} strokeWidth="1.5" strokeLinecap="round" strokeLinejoin="round" />
   379→    </svg>
   380→  );
   381→}

   483→/* =====================================================================
   484→ * 1. Occupancy heatmap (24h × 7d)
   485→ * ===================================================================== */
   486→
   487→function OccupancyHeatmap() {
   488→  const leftPad = 40;
   489→  const topPad = 22;
   490→  const rightPad = 12;
   491→  const bottomPad = 28;
   492→  const cell = 22;
   493→  const gap = 2;
   494→  const step = cell + gap;
   495→  const W = leftPad + 24 * step + rightPad;
   496→  const H = topPad + 7 * step + bottomPad;
   497→
   498→  const hours = Array.from({ length: 24 }, (_, i) => i);
   499→
   500→  return (
   501→    <ChartCard
   502→      title="Ocupación por hora"
   503→      subtitle="Intensidad de ocupación por franja horaria y día de la semana (0–100%)"
   504→    >
   505→      <div className="overflow-x-auto rp-scroll-thin -mx-1 px-1">
   506→        <svg
   507→          viewBox={`0 0 ${W} ${H}`}
   508→          className="w-full min-w-[640px] h-auto"
   509→          role="img"
   510→          aria-labelledby="heat-title heat-desc"
   511→        >
   512→          <title id="heat-title">Mapa de calor de ocupación por hora y día</title>
   513→          <desc id="heat-desc">
   514→            Matriz de 7 días por 24 horas. Mayor intensidad en franjas 13–15h y 20–22h,
   515→            especialmente viernes y sábado.
   516→          </desc>
   517→
   518→          {/* Hour labels (top) — every 3h */}
   519→          {hours.map((h) =>
   520→            h % 3 === 0 ? (
   521→              <text
   522→                key={h}
   523→                x={leftPad + h * step + cell / 2}
   524→                y={topPad - 8}
   525→                textAnchor="middle"
   526→                className="fill-muted-foreground font-mono"
   527→                fontSize="9"
   528→              >
   529→                {String(h).padStart(2, "0")}h
   530→              </text>
   531→            ) : null
   532→          )}
   533→
   534→          {/* Day labels (left) */}
   535→          {HEAT_DAYS.map((d, i) => (
   536→            <text
   537→              key={d}
   538→              x={leftPad - 8}
   539→              y={topPad + i * step + cell / 2 + 3}
   540→              textAnchor="end"
   541→              className="fill-muted-foreground font-mono"
   542→              fontSize="10"
   543→            >
   544→              {d}
   545→            </text>
   546→          ))}
   547→
   548→          {/* Cells */}
   549→          {HEAT_DAYS.map((_, di) =>
   550→            hours.map((h) => {
   551→              const v = heatFor(di, h);
   552→              const x = leftPad + h * step;
   553→              const y = topPad + di * step;
   554→              return (
   555→                <rect
   556→                  key={`${di}-${h}`}
   557→                  x={x}
   558→                  y={y}
   559→                  width={cell}
   560→                  height={cell}
   561→                  rx="2"

   604→function ChannelDonut() {
   605→  const size = 180;
   606→  const r = 60;

   617→      label: c.label,
   618→      pct: c.pct,
   619→      color: c.color,
   620→      dash,
   621→      offset: -cumulative,
   622→    });
   623→    cumulative += dash;

   626→  return (

   644→            <circle
