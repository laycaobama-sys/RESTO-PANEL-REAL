   391→/** Animated number counter — uses requestAnimationFrame, transform/opacity only. */
   392→function useAnimatedNumber(value: number, duration = 350) {
   393→  const reduce = useReducedMotion();
   394→  const [display, setDisplay] = React.useState(value);
   395→  const fromRef = React.useRef(value);
   396→  const rafRef = React.useRef<number | null>(null);
   397→
   398→  React.useEffect(() => {
   399→    if (reduce) {
   400→      setDisplay(value);
   401→      return;
   402→    }
   403→    const from = fromRef.current;
   404→    const to = value;
   405→    if (from === to) return;
   406→    const start = performance.now();
   407→    function tick(now: number) {
   408→      const t = Math.min(1, (now - start) / duration);
   409→      // easeOutCubic
   410→      const eased = 1 - Math.pow(1 - t, 3);
   411→      setDisplay(Math.round(from + (to - from) * eased));
   412→      if (t < 1) {
   413→        rafRef.current = requestAnimationFrame(tick);
   414→      } else {
   415→        fromRef.current = to;
   416→      }
   417→    }
   418→    rafRef.current = requestAnimationFrame(tick);
   419→    return () => {
   420→      if (rafRef.current) cancelAnimationFrame(rafRef.current);
   421→      fromRef.current = value;
   422→    };
   423→  }, [value, duration, reduce]);
   424→
   425→  return display;
   426→}

   735→  function onCanvasDrop(e: React.DragEvent) {
   736→    if (!editMode || !draggedId) return;
   737→    e.preventDefault();
   738→    const rect = canvasRef.current?.getBoundingClientRect();
   739→    if (!rect) return;
   740→    const t = tables.find((x) => x.id === draggedId);
   741→    if (!t) return;
   742→    const w = tableW(t);
   743→    const h = tableH(t);
   744→    const x = Math.max(
   745→      4,
   746→      Math.min(e.clientX - rect.left - w / 2, rect.width - w - 4)
   747→    );
   748→    const y = Math.max(
   749→      4,
   750→      Math.min(e.clientY - rect.top - h / 2, rect.height - h - 4)
   751→    );
   752→    const nx = Math.round(x);
   753→    const ny = Math.round(y);
   754→    setTables((ts) =>
   755→      ts.map((x2) => (x2.id === draggedId ? { ...x2, x: nx, y: ny } : x2))
   756→    );
   757→    setDragOver(false);
   758→    setDraggedId(null);
   759→    toast({
   760→      title: `${t.name} reposicionada`,
   761→      description: `x ${nx} · y ${ny}`,
   762→    });
   763→  }

   767→    | { id: string; offsetX: number; offsetY: number }

   771→  function onTableTouchStart(e: React.TouchEvent, t: RpTable) {
   772→    if (!editMode) return;
   773→    if (e.touches.length !== 1) return;
   774→    const rect = canvasRef.current?.getBoundingClientRect();
   775→    if (!rect) return;
   776→    const touch = e.touches[0];
   777→    touchDragRef.current = {
   778→      id: t.id,
   779→      offsetX: touch.clientX - rect.left - t.x,
   780→      offsetY: touch.clientY - rect.top - t.y,
   781→    };
   782→    setDraggedId(t.id);
   783→    setDragOver(true);
   784→  }

   786→  function onCanvasTouchMove(e: React.TouchEvent) {
   787→    if (!editMode || !touchDragRef.current) return;
   788→    if (e.touches.length !== 1) return;
   789→    e.preventDefault(); // stop page scroll while dragging
   790→    const rect = canvasRef.current?.getBoundingClientRect();
   791→    if (!rect) return;
   792→    const touch = e.touches[0];
   793→    const drag = touchDragRef.current;
   794→    const t = tables.find((x) => x.id === drag.id);
   795→    if (!t) return;
   796→    const w = tableW(t);
   797→    const h = tableH(t);
   798→    const x = Math.max(
   799→      4,
   800→      Math.min(touch.clientX - rect.left - drag.offsetX, rect.width - w - 4)
   801→    );
   802→    const y = Math.max(
   803→      4,
   804→      Math.min(touch.clientY - rect.top - drag.offsetY, rect.height - h - 4)
   805→    );
   806→    setTables((ts) =>
   807→      ts.map((x2) => (x2.id === drag.id ? { ...x2, x: Math.round(x), y: Math.round(y) } : x2))
   808→    );
   809→  }

   811→  function onCanvasTouchEnd() {
   812→    if (!touchDragRef.current) return;
   813→    const t = tables.find((x) => x.id === touchDragRef.current!.id);
   814→    touchDragRef.current = null;
   815→    setDraggedId(null);
   816→    setDragOver(false);
   817→    if (t) {
   818→      toast({
   819→        title: `${t.name} reposicionada`,
   820→        description: `x ${t.x} · y ${t.y}`,
   821→      });
   822→    }
